/// <summary>
/// Base conversion engine for streaming archive-to-archive repacking with archive-bomb protection.
/// Entries are streamed one by one without extracting the entire archive to disk.
/// </summary>
public abstract class ArchiveEngineBase : IConversionEngine
{
    private readonly ArchivesLimits _limits;

    /// <summary>
    /// Initializes the archive engine with a copy of the specified settings.
    /// </summary>
    /// <param name="settings">Copied archive settings.</param>
    protected ArchiveEngineBase(ArchivesSettings settings)
    {
        ArgumentNullException.ThrowIfNull(settings);
        Settings = settings;
        _limits = ArchivesLimits.From(settings);
        Encoding.RegisterProvider(CodePagesEncodingProvider.Instance);
        ZipNameEncoding = Encoding.GetEncoding(settings.ZipNameCodePage);
    }

    /// <summary>
    /// Copied settings active for this engine instance.
    /// </summary>
    protected ArchivesSettings Settings { get; }

    /// <summary>
    /// Encoding used for non-UTF-8 entry names in ZIP archives.
    /// </summary>
    protected Encoding ZipNameEncoding { get; }

    /// <inheritdoc />
    public bool SupportsOperations => false;

    /// <inheritdoc />
    public abstract IReadOnlyList<ConversionPair> Pairs { get; }

    /// <inheritdoc />
    public async Task ConvertAsync(ConversionRequest request, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        bool needsFile = RequiresSeekableFile(request.From);
        await using InputFile? file = needsFile ? await request.EnsureInputFileAsync(cancellationToken).ConfigureAwait(false) : null;
        CountingStream? counted = file is null ? new CountingStream(request.Input) : null;

        long? inputBytes;
        if (request.Input.CanSeek)
        {
            // Returns Input Length
            inputBytes = file is null
            ? request.Input.Length
            : new FileInfo(file.Path).Length;
        }
        else
        {
            // Returns FileInfo Length
            inputBytes = file is null
            ? null
            : new FileInfo(file.Path).Length;
        }

        ArchiveGuard guard = new(_limits, inputBytes, () => counted?.Count ?? 0);

        await using IArchiveReader reader = OpenReader(request, file?.Path, counted);
        await using IArchiveWriter writer = OpenWriter(request);

        HashSet<string> seenNames = new(StringComparer.Ordinal);
        long consumedCompressed = 0;

        while (await reader.NextAsync(cancellationToken).ConfigureAwait(false) is { } item)
        {
            string normalizedName = EntryNames.Normalize(item.Info.Name, item.Info.IsDirectory);
            if (!seenNames.Add(normalizedName))
            {
                throw new InvalidDataException($"В архиве две записи с одним именем «{normalizedName}»: такой архив не переносится, чтобы не потерять одну из них.");
            }

            ArchiveEntryInfo info = item.Info with { Name = normalizedName };
            guard.BeginEntry(normalizedName, info.Length);

            Stream? content = item.Content is null
                ? null
                : new GuardedReadStream(item.Content, count => guard.Unpacked(normalizedName, count));

            await writer.WriteAsync(info, content, cancellationToken).ConfigureAwait(false);

            consumedCompressed += item.CompressedLength ?? 0;
            ReportProgress(request, inputBytes, counted?.Count ?? consumedCompressed);
        }

        writer.Complete();
        request.ReportProgress(100);
    }

    /// <summary>
    /// Determines whether the input format requires a random-access file on disk (such as ZIP, 7z, or RAR).
    /// </summary>
    protected virtual bool RequiresSeekableFile(FileFormat format)
    {
        return format.Equals(FileFormat.Zip);
    }

    /// <summary>
    /// Opens an <see cref="IArchiveReader"/> for the request input.
    /// </summary>
    protected virtual IArchiveReader OpenReader(ConversionRequest request, string? filePath, Stream? countedStream)
    {
        if (request.From.Equals(FileFormat.Zip))
        {
            return OpenZipReader(filePath!);
        }

        if (request.From.Equals(FileFormat.Tar))
        {
            return new TarArchiveReader(countedStream!);
        }

        if (request.From.Equals(FileFormat.TarGz))
        {
            return new TarGzReader(new GZipStream(countedStream!, CompressionMode.Decompress));
        }

        if (request.From.Equals(FileFormat.Gzip))
        {
            return new SingleFileArchiveReader(
                new GZipStream(countedStream!, CompressionMode.Decompress),
                StripSingleArchiveExtension(request.InputPath, ".gz"),
                DateTimeOffset.UtcNow);
        }

        // Unsupported archive format
        throw new NotSupportedException($"Unsupported input archive format: {request.From.Name}.");
    }

    /// <summary>
    /// Opens an <see cref="IArchiveWriter"/> for the request target.
    /// </summary>
    protected virtual IArchiveWriter OpenWriter(ConversionRequest request)
    {
        return request.To switch
        {
            FileFormat format when format.Equals(FileFormat.Zip) =>
                new ZipArchiveWriter(request.Output, Settings.CompressionLevel),

            FileFormat format when format.Equals(FileFormat.Tar) =>
                new TarArchiveWriter(request.Output, request.TempFolder),

            FileFormat format when format.Equals(FileFormat.TarGz) =>
                new TarGzArchiveWriter(request.Output, Settings.CompressionLevel, request.TempFolder),

            FileFormat format when format.Equals(FileFormat.Gzip) =>
                new SingleFileArchiveWriter(
                    new GZipStream(request.Output, Settings.CompressionLevel, leaveOpen: true),
                    "GZip"),

            _ => throw new NotSupportedException($"Unsupported target archive format: {request.To?.Name}.")
        };
    }


    /// <summary>
    /// Opens a ZIP reader and wraps format errors into <see cref="InvalidDataException"/>.
    /// </summary>
    protected ZipArchiveReader OpenZipReader(string path)
    {
        try
        {
            return new ZipArchiveReader(path, ZipNameEncoding);
        }
        catch (InvalidDataException ex)
        {
            throw new InvalidDataException("Файл ZIP повреждён или оборван: " + ex.Message, ex);
        }
    }

    /// <summary>
    /// Derives the inner entry name for single-file compressed streams (such as .gz, .bz2, or .xz) by stripping known extensions.
    /// </summary>
    protected static string StripSingleArchiveExtension(string? inputPath, params ReadOnlySpan<string> extensions)
    {
        string name = inputPath is null ? string.Empty : Path.GetFileName(inputPath);
        foreach (string ext in extensions)
        {
            if (name.EndsWith(ext, StringComparison.OrdinalIgnoreCase))
            {
                name = name[..^ext.Length];
                break;
            }
        }

        return name.Length == 0 ? "data" : name;
    }

    private static void ReportProgress(ConversionRequest request, long? inputBytes, long consumed)
    {
        if (inputBytes is > 0)
        {
            request.ReportProgress((int)Math.Min(99, consumed * 100 / inputBytes.Value));
        }
    }
}

using GLAZOV.M.CONVERTER.Archives.Model;
using System.IO.Compression;
using System.Text;

namespace GLAZOV.M.CONVERTER.Archives.Reading;

/// <summary>
/// Reads the entries of a ZIP file. The file must be readable at any place (the central directory lies at the end), so the engine gives it as a file.
/// </summary>
public sealed class ZipArchiveReader : IArchiveReader
{
    private const int SymbolicLinkType = 0xA000;

    private readonly ZipArchive _zip;
    private readonly IEnumerator<ZipArchiveEntry> _entries;
    private Stream? _open;

    /// <summary>
    /// Opens the archive.
    /// </summary>
    /// <param name="path">The ZIP file.</param>
    /// <param name="nameEncoding">Encoding of the names of entries not marked as UTF-8.</param>
    /// <exception cref="InvalidDataException">The file is not a readable ZIP archive.</exception>
    internal ZipArchiveReader(string path, Encoding nameEncoding)
    {
        FileStream file = new(path, FileMode.Open, FileAccess.Read, FileShare.Read, 4096, FileOptions.Asynchronous | FileOptions.RandomAccess);
        try
        {
            _zip = new ZipArchive(file, ZipArchiveMode.Read, leaveOpen: false, nameEncoding);
            _entries = _zip.Entries.GetEnumerator();
        }
        catch
        {
            file.Dispose();
            throw;
        }
    }

    /// <inheritdoc />
    public async Task<ArchiveItem?> NextAsync(CancellationToken cancellationToken = default)
    {
        if (_open is not null)
        {
            await _open.DisposeAsync().ConfigureAwait(false);
            _open = null;
        }

        if (!_entries.MoveNext())
        {
            return null;
        }

        ZipArchiveEntry entry = _entries.Current;
        bool directory = entry.FullName.EndsWith('/') || entry.FullName.EndsWith('\\');
        if (entry.IsEncrypted)
        {
            throw new InvalidDataException($"Р—Р°РїРёСЃСЊ В«{entry.FullName}В» Р·Р°С€РёС„СЂРѕРІР°РЅР°: Р·Р°С€РёС„СЂРѕРІР°РЅРЅС‹Рµ Р°СЂС…РёРІС‹ РЅРµ РїРѕРґРґРµСЂР¶РёРІР°СЋС‚СЃСЏ.");
        }

/// <summary>
/// Writes a TAR archive (POSIX pax: long names and any characters are fine) to a stream, entry by entry.
/// </summary>
/// <remarks>
/// Creates the writer.
/// </remarks>
/// <param name="output">Where the archive goes; it stays open.</param>
/// <param name="tempFolder">Folder for the entries whose size is not known before they are read.</param>
public sealed class TarArchiveWriter(Stream output, string tempFolder) : IArchiveWriter
{
    private readonly TarWriter _tar = new(output, TarEntryFormat.Pax, leaveOpen: true);
    private readonly Stream _output = output;
    private readonly string _tempFolder = tempFolder;
    private int _entries;

    /// <inheritdoc />
    public async Task WriteAsync(ArchiveEntryInfo info, Stream? content, CancellationToken cancellationToken = default)
    {
        _entries++;
        if (content is null)
        {
            PaxTarEntry folder = new(TarEntryType.Directory, info.Name) { ModificationTime = info.Modified, Mode = info.Mode ?? (UnixFileMode)0b111_101_101 };
            await _tar.WriteEntryAsync(folder, cancellationToken).ConfigureAwait(false);
            return;
        }

        // TAR writes the size before the content: a stream whose size is not known goes through a temporary file first.
        string spool = Path.Combine(_tempFolder, $"archive-{Guid.NewGuid():N}.tmp");
        try
        {
            await using FileStream file = new(spool, FileMode.CreateNew, FileAccess.ReadWrite, FileShare.None, 81920, FileOptions.Asynchronous | FileOptions.DeleteOnClose);
            await content.CopyToAsync(file, cancellationToken).ConfigureAwait(false);
            file.Position = 0;
            PaxTarEntry entry = new(TarEntryType.RegularFile, info.Name)
            {
                ModificationTime = info.Modified,
                Mode = info.Mode ?? (UnixFileMode)0b110_100_100,
                DataStream = file,
            };
            await _tar.WriteEntryAsync(entry, cancellationToken).ConfigureAwait(false);
        }
        finally
        {
            if (File.Exists(spool))
            {
                File.Delete(spool);
            }
        }
    }

    /// <inheritdoc />
    public void Complete()
    {
    }

    /// <inheritdoc />
    public async ValueTask DisposeAsync()
    {
        await _tar.DisposeAsync().ConfigureAwait(false);
        if (_entries == 0)
        {
            // TarWriter writes nothing when it has no entries; an empty TAR archive is two blocks of zeroes.
            await _output.WriteAsync(new byte[1024]).ConfigureAwait(false);

/// <summary>
/// Asks the TAR detector of the Formats package whether the first block of an unpacked stream is a TAR header.
/// </summary>
internal static class TarProbe
{
    private const int BlockLength = 512;

    /// <summary>
    /// True when the first 512 bytes of the stream are a TAR header.
    /// </summary>
    /// <param name="unpacked">The unpacked stream, read from its start.</param>
    internal static bool Matches(Stream unpacked)
    {
        byte[] block = new byte[BlockLength];
        int total = 0;
        int read;
        while (total < block.Length && (read = unpacked.Read(block, total, block.Length - total)) > 0)
        {
            total += read;
        }

        return total == block.Length && StandardDetectors.Archives.Detect(block)?.Equals(FileFormat.Tar) == true;
    }
}

/// <summary>
/// Writes a TAR archive compressed with BZip2 to a stream.
/// </summary>
internal sealed class TarBz2ArchiveWriter : IArchiveWriter
{
    private readonly BZip2Stream _bzip2;
    private readonly TarArchiveWriter _tar;

    /// <summary>
    /// Creates the writer.
    /// </summary>
    /// <param name="output">Where the archive goes; it stays open.</param>
    /// <param name="tempFolder">Folder for temporary files.</param>
    internal TarBz2ArchiveWriter(Stream output, string tempFolder)
    {
        _bzip2 = new BZip2Stream(new KeepOpenStream(output), CompressionMode.Compress, false);
        _tar = new TarArchiveWriter(_bzip2, tempFolder);
    }

    /// <inheritdoc />
    public Task WriteAsync(ArchiveEntryInfo info, Stream? content, CancellationToken cancellationToken)
    {
        return _tar.WriteAsync(info, content, cancellationToken);
    }

    /// <inheritdoc />
    public void Complete()
    {
        _tar.Complete();
    }

    /// <inheritdoc />
    public async ValueTask DisposeAsync()
    {
        await _tar.DisposeAsync().ConfigureAwait(false);
        await _bzip2.DisposeAsync().ConfigureAwait(false);
    }
