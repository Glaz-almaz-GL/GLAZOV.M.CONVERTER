    public IReadOnlyList<IConversionEngine> Engines { get; } = [];
}

using GLAZOV.M.CONVERTER.Data;
using GLAZOV.M.CONVERTER.Extensions;

namespace GLAZOV.M.CONVERTER.Formats.Detection;

/// <summary>
/// A detector of one family of formats (audio, images, video, documents): its signs are tried in the order of the list, the first that fits names the format.
/// </summary>
/// <param name="signatures">Formats with the tests of their signs, in the order of asking.</param>
internal abstract class FamilyDetector((FileFormat Format, SignatureTest Matches)[] signatures) : IFormatDetector
{
    /// <inheritdoc />
    public virtual int HeadLength => DetectionSizes.MinimumWindow;

    /// <inheritdoc />
    public FileFormat? Detect(ReadOnlySpan<byte> head)
    {
        foreach ((FileFormat format, SignatureTest matches) in signatures)
        {
            if (matches(head))
            {
                return format;
            }
        }

        return null;
    }
}

using GLAZOV.M.CONVERTER.Data;
using GLAZOV.M.CONVERTER.Extensions;
using GLAZOV.M.CONVERTER.Formats.Detection.Documents;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Archives;

/// <summary>
/// Archives: TAR.GZ, GZip, ZIP, TAR. A ZIP file that holds a document (DOCX, ODT and the like) is left to the document detector; when the start of
/// the input does not settle it, the second look reads the central directory and names ZIP only if no document format fits.
/// </summary>
internal sealed class ArchiveDetector() : FamilyDetector(
[
    (FileFormat.TarGz, TarGzSignature.Matches),
    (FileFormat.Gzip, GzipSignature.Matches),
    (FileFormat.Zip, ZipSignature.Matches),
    (FileFormat.Tar, TarSignature.Matches),
]), IStreamFormatDetector
{
    /// <inheritdoc />
    public bool CanInspect(ReadOnlySpan<byte> head)
    {
        return head.StartsWith(ZipPackage.ZipMagic);
    }

    /// <inheritdoc />
    public FileFormat? Detect(Stream input)
    {
        if (ContainerProbe.Match(input) is not null)
        {
            return null;
        }

        try
        {
            input.Position = 0;
            using System.IO.Compression.ZipArchive zip = new(input, System.IO.Compression.ZipArchiveMode.Read, leaveOpen: true);
            return zip.Entries.Count > 0 ? FileFormat.Zip : null;
        }
        catch (Exception ex) when (ex is InvalidDataException or IOException or NotSupportedException or ArgumentException)
        {
            return null;
        }
    }
}

/// <summary>
/// TAR.GZ: a GZip file whose first 512 unpacked bytes are a TAR header. The window holds the compressed start, which is
/// enough because a header packs to far less than the window.
/// </summary>
internal static class TarGzSignature
{
    /// <summary>
    /// True when the data starts like this format.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        if (head.Length < GzipSignature.HeaderLength || !head.StartsWith(new byte[] { 0x1F, 0x8B, 0x08 }))
        {
            return false;
        }

        try
        {
            using MemoryStream packed = new(head.ToArray());
            using GZipStream unpacked = new(packed, CompressionMode.Decompress);
            byte[] header = new byte[TarSignature.BlockLength];
            int total = 0;
            int read;
            while (total < header.Length && (read = unpacked.Read(header, total, header.Length - total)) > 0)
            {
                total += read;
            }

            return total == header.Length && TarSignature.Matches(header);
        }
        catch (Exception ex) when (ex is InvalidDataException or IOException or EndOfStreamException)
        {
            return false;
        }
    }

/// <summary>
/// TAR: the checksum of the first 512-byte block as it is written in the block, together with "ustar" at byte 257 (POSIX and GNU)
/// or, for the old format without any magic, an octal mode and size, a name and a plain type flag. The checksum alone is not
/// taken for a sign: a block of zeroes or of random bytes does not pass it.
/// </summary>
internal static class TarSignature
{
    internal const int BlockLength = 512;

    private const int MagicOffset = 257;
    private const int ChecksumOffset = 148;
    private const int ChecksumLength = 8;
    private const int ModeOffset = 100;
    private const int SizeOffset = 124;
    private const int TypeFlagOffset = 156;

    /// <summary>
    /// True when the data starts like this format.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        return head.Length >= BlockLength && ReadOctal(head.Slice(ChecksumOffset, ChecksumLength)) is { } written && written == Sum(head[..BlockLength]) && (head.Slice(MagicOffset, 5).SequenceEqual("ustar"u8) || LooksLikeOldHeader(head));
    }

    // The old format has no magic: a name, an octal mode (bytes 100-107) and size (124-135), a type flag that is 0 or "0"-"7".
    private static bool LooksLikeOldHeader(ReadOnlySpan<byte> head)
    {
        byte type = head[TypeFlagOffset];
        return head[0] != 0
            && ReadOctal(head.Slice(ModeOffset, 8)) is not null
            && ReadOctal(head.Slice(SizeOffset, 12)) is not null
            && (type is 0 || type is >= (byte)'0' and <= (byte)'7');
    }

    // The checksum field counts as eight spaces while the sum is taken.
    private static long Sum(ReadOnlySpan<byte> block)
    {
        long sum = 0;
        for (int i = 0; i < block.Length; i++)
        {
            if (i is >= ChecksumOffset and < (ChecksumOffset + ChecksumLength))
            {
                sum += (byte)' ';
            }
            else
            {
                sum += block[i];
            }
        }

        return sum;
    }

    // Octal digits, then NUL or a space; nothing else is a number.
    private static long? ReadOctal(ReadOnlySpan<byte> field)
    {
        long value = 0;
        int digits = 0;
        foreach (byte b in field)
        {
            if (b is >= (byte)'0' and <= (byte)'7')
            {
                value = (value * 8) + (b - '0');
                digits++;
            }
            else if (b is 0 or (byte)' ')
            {
                if (digits > 0)
                {
                    return value;
                }
            }
            else
            {
                return null;
            }
        }

        return digits > 0 ? value : null;
    }
}

namespace GLAZOV.M.CONVERTER.Formats.Detection.Archives;

/// <summary>
/// ZIP: a local file header ("PK 03 04") or an empty archive ("PK 05 06") that is not the package of a document. The start of a big
/// archive cannot tell a package of a document from a plain archive (the parts that name a package may lie further), so by the start alone
/// ZIP is named only when the whole file lies inside the window and shows no part of a document, or the first entry already rules a document out
/// (an entry "mimetype" that is not OpenDocument, as in EPUB; "META-INF/", as in JAR). Everything else is left to the second look,
/// which reads the central directory; an input that cannot be read back gets no answer.
/// </summary>
internal static class ZipSignature
{
    private const int HeaderLength = 30;
    private const int EmptyArchiveLength = 22;
    private const string OdfMimeStart = "application/vnd.oasis.opendocument";

    private static readonly byte[] EmptyMagic = [0x50, 0x4B, 0x05, 0x06];

    private static readonly string[] NotDocumentStarts = ["META-INF/"];

    private static readonly string[] DocumentStarts = ["[Content_Types].xml", "_rels/", "docProps/", "customXml/", "word/", "xl/", "ppt/"];

    /// <summary>
    /// True when the data starts like this format.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        if (head.StartsWith(EmptyMagic))
        {
            return head.Length >= EmptyArchiveLength;
        }

        if (!head.StartsWith(ZipPackage.ZipMagic) || head.Length < HeaderLength || ZipPackage.Kind(head) is not null)
        {
            return false;
        }

        ushort nameLength = System.Buffers.Binary.BinaryPrimitives.ReadUInt16LittleEndian(head[26..28]);
        if (nameLength == 0 || HeaderLength + nameLength > head.Length)
        {
            return false;
        }

        string first = System.Text.Encoding.UTF8.GetString(head.Slice(HeaderLength, nameLength));
        if (first == "mimetype")
        {
            return !StartsOdf(head, nameLength);
        }

        if (NotDocumentStarts.Any(start => first.StartsWith(start, StringComparison.Ordinal)))
        {
            return true;
        }

        if (DocumentStarts.Any(start => first.StartsWith(start, StringComparison.Ordinal)))
        {
            return false;
        }

        (bool contentTypes, HashSet<Data.FileFormat> mainParts, HashSet<Data.FileFormat> folders) = ZipPackage.Scan(head);
        return !contentTypes && mainParts.Count == 0 && folders.Count == 0 && IsWhole(head);
    }

    // The end-of-central-directory record ("PK 05 06", a comment that reaches exactly the end of the data) shows the window holds the whole file.
    private static bool IsWhole(ReadOnlySpan<byte> head)
    {
        int lowest = Math.Max(0, head.Length - (EmptyArchiveLength + ushort.MaxValue));
        for (int at = head.Length - EmptyArchiveLength; at >= lowest; at--)
        {
            if (head[at..].StartsWith(EmptyMagic) && at + EmptyArchiveLength + System.Buffers.Binary.BinaryPrimitives.ReadUInt16LittleEndian(head.Slice(at + 20, 2)) == head.Length)
            {
                return true;
            }
        }

        return false;
    }

    // The "mimetype" entry is stored first; an archive whose entry is cut off by the window is left to the second look.
    private static bool StartsOdf(ReadOnlySpan<byte> head, int nameLength)
    {
        ushort extraLength = System.Buffers.Binary.BinaryPrimitives.ReadUInt16LittleEndian(head[28..30]);
        int dataStart = HeaderLength + nameLength + extraLength;
        uint size = System.Buffers.Binary.BinaryPrimitives.ReadUInt32LittleEndian(head[18..22]);
        if (dataStart + size > head.Length || size > 100)
        {
            return true;
        }

        return System.Text.Encoding.ASCII.GetString(head.Slice(dataStart, (int)size)).StartsWith(OdfMimeStart, StringComparison.Ordinal);
    }
}

/// <summary>
/// MP3: an ID3v2 tag, or a frame of layer III followed exactly where its length promises by a second frame of the same version.
/// </summary>
internal static class Mp3Signature
{
    private static readonly int[] Mpeg1Layer3Bitrates = [0, 32, 40, 48, 56, 64, 80, 96, 112, 128, 160, 192, 224, 256, 320];
    private static readonly int[] Mpeg2Layer3Bitrates = [0, 8, 16, 24, 32, 40, 48, 56, 64, 80, 96, 112, 128, 144, 160];
    private static readonly int[] Mpeg1SampleRates = [44100, 48000, 32000];
    private static readonly int[] Mpeg2SampleRates = [22050, 24000, 16000];
    private static readonly int[] Mpeg25SampleRates = [11025, 12000, 8000];

    /// <summary>
    /// True when the data starts like MP3.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        // The ID3v2 tag: "ID3", version 2-4 (the next byte of the version is not 0xFF), size bytes of 7 bits each.
        if (head.Length >= 10 && head.StartsWith("ID3"u8)
            && head[3] is >= 2 and <= 4 && head[4] != 0xFF
            && head[6] < 0x80 && head[7] < 0x80 && head[8] < 0x80 && head[9] < 0x80)
        {
            return true;
        }

        // Without a tag it is a frame of layer III; so that random data does not pass for MP3,
        // a second frame is required exactly where the length of the first promises it.
        if (!TryReadFrame(head, out Frame first))
        {
            return false;
        }

        int next = first.Length;
        if (head.Length < next + 4 || !TryReadFrame(head[next..], out Frame second))
        {
            return false;
        }

        return second.Version == first.Version;
    }

    private static bool TryReadFrame(ReadOnlySpan<byte> data, out Frame frame)
    {
        frame = default;
        if (data.Length < 4 || data[0] != 0xFF || (data[1] & 0xE0) != 0xE0)
        {
            return false;
        }

        int versionBits = (data[1] >> 3) & 0b11;     // 00 - MPEG 2.5, 01 - forbidden, 10 - MPEG 2, 11 - MPEG 1
        int layerBits = (data[1] >> 1) & 0b11;       // 01 - layer III
        int bitrateIndex = data[2] >> 4;
        int sampleRateIndex = (data[2] >> 2) & 0b11;
        int padding = (data[2] >> 1) & 1;

        if (versionBits == 0b01 || layerBits != 0b01 || bitrateIndex is 0 or 0xF || sampleRateIndex == 0b11)
        {
            return false;
        }

        bool isMpeg1 = versionBits == 0b11;
        int bitrate = (isMpeg1 ? Mpeg1Layer3Bitrates : Mpeg2Layer3Bitrates)[bitrateIndex] * 1000;
        int sampleRate = (versionBits switch
        {
            0b11 => Mpeg1SampleRates,
            0b10 => Mpeg2SampleRates,
            _ => Mpeg25SampleRates,
        })[sampleRateIndex];

        int length = ((isMpeg1 ? 144 : 72) * bitrate / sampleRate) + padding;
        frame = new Frame(versionBits, length);
        return true;
    }

    private readonly record struct Frame(int Version, int Length);
}

using System.Buffers.Binary;
using System.Text;
using GLAZOV.M.CONVERTER.Data;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Documents;

/// <summary>
/// What a compound file (OLE2) holds that names its document format: DOC, XLS, PPT by the streams of the root of its directory.
/// Never throws on short or foreign data.
/// </summary>
internal static class CompoundFile
{
    internal static readonly byte[] CompoundMagic = [0xD0, 0xCF, 0x11, 0xE0, 0xA1, 0xB1, 0x1A, 0xE1];

    internal const int CompoundHeaderLength = 512;

    internal const int CompoundEntryLength = 128;

    internal const int MaxDirectoryEntries = 4096;

    private const byte StreamObject = 2;

    /// <summary>
    /// Returns the document format of a compound file the data starts like, or <c>null</c>.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static FileFormat? Kind(ReadOnlySpan<byte> head)
    {
        return head.StartsWith(CompoundMagic) ? MatchCompound(head) : null;
    }

    // Compound file: sector size and the first directory sector are in the header; the directory entries (128 bytes each, numbered
    // from 0, entry 0 is the root) form a tree. Only the direct children of the root decide: a stream with the same name inside a
    // nested storage (an embedded object) does not. The entries read are the ones that follow the first directory sector inside
    // the window, which is how a small file lays its directory out.
    private static FileFormat? MatchCompound(ReadOnlySpan<byte> head)
    {
        if (head.Length < CompoundHeaderLength)
        {
            return null;
        }

        ushort shift = BinaryPrimitives.ReadUInt16LittleEndian(head[0x1E..0x20]);
        uint firstDirectorySector = BinaryPrimitives.ReadUInt32LittleEndian(head[0x30..0x34]);
        if (shift is not (9 or 12))
        {
            return null;
        }

        long offset = ((long)firstDirectorySector + 1) << shift;
        if (offset + CompoundEntryLength > head.Length)
        {
            return null;
        }

        return MatchDirectory(head[(int)offset..]);
    }

    /// <summary>
    /// Names the old Office format by the streams that are direct children of the root of a compound file directory.
    /// </summary>
    /// <param name="directory">Directory entries from entry 0 (the root) on.</param>
    internal static FileFormat? MatchDirectory(ReadOnlySpan<byte> directory)
    {
        int count = Math.Min(directory.Length / CompoundEntryLength, MaxDirectoryEntries);
        if (count == 0)
        {
            return null;
        }

        HashSet<string> streams = [];
        Stack<uint> pending = new();
        HashSet<uint> seen = [];
        pending.Push(Child(directory, 0));
        while (pending.Count > 0)
        {
            uint id = pending.Pop();
            if (id >= count || !seen.Add(id))
            {
                continue;
            }

            ReadOnlySpan<byte> entry = directory.Slice((int)id * CompoundEntryLength, CompoundEntryLength);
            if (entry[0x42] == StreamObject)
            {
                streams.Add(EntryName(entry));
            }

            pending.Push(BinaryPrimitives.ReadUInt32LittleEndian(entry[0x44..0x48]));
            pending.Push(BinaryPrimitives.ReadUInt32LittleEndian(entry[0x48..0x4C]));
        }

        if (streams.Contains("WordDocument"))
        {
            return FileFormat.Doc;
        }

        if (streams.Contains("PowerPoint Document"))
        {
            return FileFormat.Ppt;
        }

        return streams.Contains("Workbook") || streams.Contains("Book") ? FileFormat.Xls : null;
    }

    private static uint Child(ReadOnlySpan<byte> directory, int id)
    {
        return BinaryPrimitives.ReadUInt32LittleEndian(directory.Slice((id * CompoundEntryLength) + 0x4C, 4));
    }

    // Name of a directory entry: UTF-16 in the first 64 bytes, its length (with the end mark) at 0x40.
    private static string EntryName(ReadOnlySpan<byte> entry)
    {
        int length = BinaryPrimitives.ReadUInt16LittleEndian(entry[0x40..0x42]);
        return length is >= 2 and <= 64 ? Encoding.Unicode.GetString(entry[..(length - 2)]) : string.Empty;
    }
}

using GLAZOV.M.CONVERTER.Data;
using System.Buffers.Binary;
using System.IO.Compression;
using System.Text;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Documents;

/// <summary>
/// Second look at the containers of documents, for an input that can be read at any place: an archive by its central directory
/// (at the end of the file) and a compound file by the chain of its directory sectors, wherever they lie. Never throws on short or
/// foreign data; what cannot be read inside the budget is not guessed.
/// </summary>
internal static class ContainerProbe
{
    private const string OdfMimeType = "mimetype";
    private const int MaxOdfMimeLength = 100;
    private const int HeaderTableEntries = 109;
    private const int MagicLength = 8;
    private const uint EndOfChain = 0xFFFFFFFE;

    /// <summary>
    /// Whether the data starts like a container the probe can open.
    /// </summary>
    /// <param name="head">Leading bytes of the input.</param>
    internal static bool IsContainer(ReadOnlySpan<byte> head)
    {
        return head.StartsWith(ZipPackage.ZipMagic) || head.StartsWith(CompoundFile.CompoundMagic);
    }

    /// <summary>
    /// Returns the document format of the container, or <c>null</c>.
    /// </summary>
    /// <param name="stream">The input; position 0 is its start.</param>
    internal static FileFormat? Match(Stream stream)
    {
        try
        {
            byte[] start = new byte[MagicLength];
            stream.Position = 0;
            int read = stream.Read(start, 0, start.Length);
            if (read == start.Length && start.AsSpan().StartsWith(CompoundFile.CompoundMagic))
            {
                // Compound matching
                return MatchCompound(stream);
            }

            return read >= ZipPackage.ZipMagic.Length && start.AsSpan().StartsWith(ZipPackage.ZipMagic)
                ? MatchArchive(stream)
                : null;
        }
        catch (Exception ex) when (ex is InvalidDataException or IOException or NotSupportedException or ArgumentException)
        {
            return null;
        }
    }

    private static FileFormat? MatchArchive(Stream stream)
    {
        using ZipArchive zip = new(stream, ZipArchiveMode.Read, leaveOpen: true);
        if (zip.Entries.Count > 0 && zip.Entries[0].FullName == OdfMimeType)
        {
            return MatchOdf(zip.Entries[0]);
        }

        bool contentTypes = false;
        HashSet<FileFormat> mainParts = [];
        HashSet<FileFormat> folders = [];
        foreach (string entryFullName in zip.Entries.Select(entry => entry.FullName))
        {
            contentTypes |= entryFullName == ZipPackage.ContentTypesPart;
            if (ZipPackage.MainPart(entryFullName) is { } main)
            {
                mainParts.Add(main);
            }

            if (ZipPackage.Folder(entryFullName) is { } folder)
            {
                folders.Add(folder);
            }
        }

        return ZipPackage.ChooseOffice(contentTypes, mainParts, folders);
    }

    private static FileFormat? MatchOdf(ZipArchiveEntry entry)
    {
        if (entry.Length > MaxOdfMimeLength)
        {
            return null;
        }

        using Stream content = entry.Open();
        byte[] buffer = new byte[MaxOdfMimeLength];
        int total = 0;
        int read;
        while (total < buffer.Length && (read = content.Read(buffer, total, buffer.Length - total)) > 0)
        {
            total += read;
        }

        return ZipPackage.OdfFormat(Encoding.ASCII.GetString(buffer, 0, total));
    }

    // The header names the first directory sector; the sector allocation table (found through the header and the chain of extra
    // table sectors) gives the next sector of the directory. Only the table sectors that are needed are read.
    private static FileFormat? MatchCompound(Stream stream)
    {
        byte[] header = new byte[CompoundFile.CompoundHeaderLength];
        stream.Position = 0;
        if (!ReadFully(stream, header))
        {
            return null;
        }

        int shift = BinaryPrimitives.ReadUInt16LittleEndian(header.AsSpan(0x1E, 2));
        if (shift is not (9 or 12))
        {
            return null;
        }

        int sectorSize = 1 << shift;
        uint tableCount = BinaryPrimitives.ReadUInt32LittleEndian(header.AsSpan(0x2C, 4));
        uint firstDirectory = BinaryPrimitives.ReadUInt32LittleEndian(header.AsSpan(0x30, 4));
        uint firstExtra = BinaryPrimitives.ReadUInt32LittleEndian(header.AsSpan(0x44, 4));
        uint extraCount = BinaryPrimitives.ReadUInt32LittleEndian(header.AsSpan(0x48, 4));

        List<uint> tables = ReadTableSectors(stream, header, sectorSize, tableCount, firstExtra, extraCount);
        if (tables.Count == 0)
        {
            return null;
        }

        byte[]? directory = ReadDirectory(stream, firstDirectory, sectorSize, tables);
        return directory is null ? null : CompoundFile.MatchDirectory(directory);
    }

    private static List<uint> ReadTableSectors(Stream stream, byte[] header, int sectorSize, uint tableCount, uint firstExtra, uint extraCount)
    {
        List<uint> tables = [];
        for (int i = 0; i < HeaderTableEntries && tables.Count < tableCount; i++)
        {
            tables.Add(BinaryPrimitives.ReadUInt32LittleEndian(header.AsSpan(0x4C + (i * 4), 4)));
        }

        byte[] sector = new byte[sectorSize];
        int entries = (sectorSize / 4) - 1;
        uint next = firstExtra;
        for (uint i = 0; i < extraCount && tables.Count < tableCount && next < EndOfChain; i++)
        {
            if (!ReadSector(stream, next, sector))
            {
                break;
            }

            for (int e = 0; e < entries && tables.Count < tableCount; e++)
            {
                tables.Add(BinaryPrimitives.ReadUInt32LittleEndian(sector.AsSpan(e * 4, 4)));
            }

            next = BinaryPrimitives.ReadUInt32LittleEndian(sector.AsSpan(entries * 4, 4));
        }

        return tables;
    }

    private static byte[]? ReadDirectory(Stream stream, uint first, int sectorSize, List<uint> tables)
    {
        int perTable = sectorSize / 4;
        int maxSectors = Math.Max(1, CompoundFile.MaxDirectoryEntries * CompoundFile.CompoundEntryLength / sectorSize);
        Dictionary<uint, byte[]> loaded = [];
        HashSet<uint> seen = [];
        using MemoryStream directory = new();
        byte[] sector = new byte[sectorSize];
        uint current = first;
        while (current < EndOfChain && seen.Count < maxSectors && seen.Add(current))
        {
            if (!ReadSector(stream, current, sector))
            {
                return null;
            }

            directory.Write(sector);

            uint tableIndex = current / (uint)perTable;
            if (tableIndex >= tables.Count)
            {
                break;
            }

            uint table = tables[(int)tableIndex];
            if (!loaded.TryGetValue(table, out byte[]? entries))
            {
                entries = new byte[sectorSize];
                if (!ReadSector(stream, table, entries))
                {
                    return null;
                }

                loaded[table] = entries;
            }

            current = BinaryPrimitives.ReadUInt32LittleEndian(entries.AsSpan((int)(current % (uint)perTable) * 4, 4));
        }

        return directory.Length == 0 ? null : directory.ToArray();
    }

    // Sector n starts after the header sector, at (n + 1) << shift.
    private static bool ReadSector(Stream stream, uint id, byte[] sector)
    {
        if (id >= EndOfChain)
        {
            return false;
        }

        long offset = ((long)id + 1) * sector.Length;
        if (offset + sector.Length > stream.Length)
        {
            return false;
        }

        stream.Position = offset;
        return ReadFully(stream, sector);
    }

    private static bool ReadFully(Stream stream, byte[] buffer)
    {
        int total = 0;
        while (total < buffer.Length)
        {
            int read = stream.Read(buffer, total, buffer.Length - total);
            if (read == 0)
            {
                return false;
            }

            total += read;
        }

        return true;
    }
}

using System.Buffers.Binary;
using System.Text;
using GLAZOV.M.CONVERTER.Data;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Documents;

/// <summary>
/// What an archive (ZIP) holds that names its document format: ODT, ODS, ODP by the first entry "mimetype", DOCX, XLSX, PPTX by the entries
/// that lie inside the detection window. Never throws on short or foreign data; what cannot be confirmed inside the window is not guessed.
/// </summary>
internal static class ZipPackage
{
    internal static readonly byte[] ZipMagic = [0x50, 0x4B, 0x03, 0x04];

    private static readonly byte[] DescriptorMagic = [0x50, 0x4B, 0x07, 0x08];

    private const int ZipHeaderLength = 30;

    // The folder of an OOXML package names its kind; the entries come in no fixed order, so any entry inside the window shows it.
    internal const string ContentTypesPart = "[Content_Types].xml";

    private const string WordFolder = "word/";

    private const string ExcelFolder = "xl/";

    private const string PowerPointFolder = "ppt/";

    /// <summary>
    /// Returns the document format of an archive the data starts like, or <c>null</c>.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static FileFormat? Kind(ReadOnlySpan<byte> head)
    {
        return head.StartsWith(ZipMagic) ? MatchArchive(head) : null;
    }

    // ODF has "mimetype" first and stored. Other archives are searched for local file headers inside the window: a writer that
    // streams its output hides the entry sizes (data descriptor), so the headers cannot be walked, but each one starts with the magic
    // and carries the name. An Office Open XML package is confirmed by its main part, or by the content types entry together
    // with an entry of the folder of one kind: a plain archive with a folder called "word/" is not a document.
    private static FileFormat? MatchArchive(ReadOnlySpan<byte> head)
    {
        FileFormat? odf = MatchOdf(head);
        if (odf is not null)
        {
            return odf;
        }

        (bool contentTypes, HashSet<FileFormat> mainParts, HashSet<FileFormat> folders) = Scan(head);
        return ChooseOffice(contentTypes, mainParts, folders);
    }

    /// <summary>
    /// Walks the local file headers inside the window and reports what they name: the content types entry, the main parts and the folders of Office Open XML.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static (bool ContentTypes, HashSet<FileFormat> MainParts, HashSet<FileFormat> Folders) Scan(ReadOnlySpan<byte> head)
    {
        bool contentTypes = false;
        HashSet<FileFormat> mainParts = [];
        HashSet<FileFormat> folders = [];
        int offset = 0;
        while (offset + ZipHeaderLength <= head.Length && head[offset..].StartsWith(ZipMagic))
        {
            ReadOnlySpan<byte> header = head[offset..];
            ushort flags = BinaryPrimitives.ReadUInt16LittleEndian(header[6..8]);
            uint compressed = BinaryPrimitives.ReadUInt32LittleEndian(header[18..22]);
            ushort nameLength = BinaryPrimitives.ReadUInt16LittleEndian(header[26..28]);
            ushort extraLength = BinaryPrimitives.ReadUInt16LittleEndian(header[28..30]);
            int dataStart = ZipHeaderLength + nameLength + extraLength;
            if (nameLength == 0 || offset + dataStart > head.Length)
            {
                break;
            }

            string name = Encoding.UTF8.GetString(header.Slice(ZipHeaderLength, nameLength));
            contentTypes |= name == ContentTypesPart;
            if (MainPart(name) is { } main)
            {
                mainParts.Add(main);
            }

            if (Folder(name) is { } folder)
            {
                folders.Add(folder);
            }

            if ((flags & 0x08) == 0)
            {
                // The size is known: skip the data, so an archive stored inside is not read as part of this one.
                if (compressed == uint.MaxValue || (long)offset + dataStart + compressed > head.Length)
                {
                    break;
                }

                offset += dataStart + (int)compressed;
                continue;
            }

            // A data descriptor hides the size of streamed data. The end of the data is found by the descriptor itself: its signature
            // followed by the compressed size, which must equal the distance from the start of the data. Bytes of an archive stored
            // inside carry descriptors of their own whose sizes do not fit this distance, so they are not mistaken for ours.
            int next = FindAfterDescriptor(header, dataStart);
            if (next < 0)
            {
                break;
            }

            offset += next;
        }
        return (contentTypes, mainParts, folders);
    }

    /// <summary>
    /// Decides the Office Open XML kind from what an archive holds: one main part, or the content types entry with one kind of folder.
    /// </summary>
    internal static FileFormat? ChooseOffice(bool contentTypes, HashSet<FileFormat> mainParts, HashSet<FileFormat> folders)
    {
        if (mainParts.Count == 1)
        {
            return mainParts.Single();
        }

        return contentTypes && mainParts.Count == 0 && folders.Count == 1 ? folders.Single() : null;
    }

    // Offset (from the header) of the next local header after the data descriptor of a streamed entry, or -1.
    private static int FindAfterDescriptor(ReadOnlySpan<byte> header, int dataStart)
    {
        int from = dataStart;
        while (from < header.Length)
        {
            int at = header[from..].IndexOf(DescriptorMagic);
            if (at < 0)
            {
                return -1;
            }

            int descriptor = from + at;
            long distance = descriptor - dataStart;
            if (descriptor + 16 <= header.Length && BinaryPrimitives.ReadUInt32LittleEndian(header.Slice(descriptor + 8, 4)) == distance
                && header[(descriptor + 16)..].StartsWith(ZipMagic))
            {
                return descriptor + 16;
            }

            if (descriptor + 24 <= header.Length && BinaryPrimitives.ReadUInt64LittleEndian(header.Slice(descriptor + 8, 8)) == (ulong)distance
                && header[(descriptor + 24)..].StartsWith(ZipMagic))
            {
                return descriptor + 24;
            }

            from = descriptor + DescriptorMagic.Length;
        }

        return -1;
    }

    private static FileFormat? MatchOdf(ReadOnlySpan<byte> head)
    {
        if (head.Length < ZipHeaderLength)
        {
            return null;
        }

        ushort flags = BinaryPrimitives.ReadUInt16LittleEndian(head[6..8]);
        uint size = BinaryPrimitives.ReadUInt32LittleEndian(head[18..22]);
        ushort nameLength = BinaryPrimitives.ReadUInt16LittleEndian(head[26..28]);
        ushort extraLength = BinaryPrimitives.ReadUInt16LittleEndian(head[28..30]);
        int dataStart = ZipHeaderLength + nameLength + extraLength;
        if ((flags & 0x08) != 0 || dataStart > head.Length || !head.Slice(ZipHeaderLength, nameLength).SequenceEqual("mimetype"u8))
        {
            return null;
        }

        return MatchMimeType(head, dataStart, size);
    }

    internal static FileFormat? MainPart(string name)
    {
        return name switch
        {
            "word/document.xml" => FileFormat.Docx,
            "xl/workbook.xml" => FileFormat.Xlsx,
            "ppt/presentation.xml" => FileFormat.Pptx,
            _ => null,
        };
    }

    internal static FileFormat? Folder(string name)
    {
        if (name.StartsWith(WordFolder, StringComparison.Ordinal))
        {
            return FileFormat.Docx;
        }

        if (name.StartsWith(ExcelFolder, StringComparison.Ordinal))
        {
            return FileFormat.Xlsx;
        }

        return name.StartsWith(PowerPointFolder, StringComparison.Ordinal) ? FileFormat.Pptx : null;
    }

    private static FileFormat? MatchMimeType(ReadOnlySpan<byte> header, int dataStart, uint size)
    {
        if (dataStart + size > header.Length || size > 100)
        {
            return null;
        }

        return OdfFormat(Encoding.ASCII.GetString(header.Slice(dataStart, (int)size)));
    }

    /// <summary>
    /// Maps the content of the "mimetype" entry of an OpenDocument package to a format.
    /// </summary>
    internal static FileFormat? OdfFormat(string mime)
    {
        return mime switch
        {
            "application/vnd.oasis.opendocument.text" => FileFormat.Odt,
            "application/vnd.oasis.opendocument.spreadsheet" => FileFormat.Ods,
            "application/vnd.oasis.opendocument.presentation" => FileFormat.Odp,
            _ => null,
        };
    }
}

/// a non-empty body that starts after the directory. CUR files ("00 00 02 00") are not ICO.
/// </summary>

{
    private const int HeaderSize = 6;
    private const int EntrySize = 16;

    /// <summary>
    /// True when the data starts like this format.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)

        {
            return false;
        }

        if (count == 0)
        {
            return false;
        }

        uint directoryEnd = (uint)(HeaderSize + (count * EntrySize));
        for (int index = 0; index < count && HeaderSize + ((index + 1) * EntrySize) <= head.Length; index++)
        {

            {
                return false;
            }
        }

using System.Text;
using GLAZOV.M.CONVERTER.Formats.Detection.Text;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Tables;

/// <summary>
/// Reads the start of a text as rows of fields: a delimiter, a quote or a line break inside quotes does not end a field or a row.
/// The signs of CSV and TSV share it.
/// </summary>
internal static class DelimitedText
{
    private const int MaxRows = 64;

    // Two lines with the same number of commas are common in prose; three are a table far more often.
    private const int MinRows = 3;

    /// <summary>
    /// True when the head reads as text of at least three non-empty rows that all hold the same non-zero number of the delimiter outside quotes.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    /// <param name="delimiters">Delimiters to try in order; the first that fits is the one used.</param>
    /// <param name="fitting">The delimiter that fits, when the result is true.</param>
    internal static bool TryFind(ReadOnlySpan<byte> head, ReadOnlySpan<char> delimiters, out char fitting)
    {
        fitting = '\0';
        if (!TextSignature.Matches(head))
        {
            return false;
        }

        string text = Decode(head);
        if (LooksLikeMarkup(text))
        {
            return false;
        }

        // A window cut mid-row leaves a last row that proves nothing; a whole input without a final line break keeps it.
        bool windowCut = head.Length >= DetectionSizes.MinimumWindow;
        foreach (char delimiter in delimiters)
        {
            if (HasEqualCounts(text, delimiter, windowCut))
            {
                fitting = delimiter;
                return true;
            }
        }

        return false;
    }

    private static string Decode(ReadOnlySpan<byte> head)
    {
        if (head.Length >= 2 && head[0] == 0xFF && head[1] == 0xFE)
        {
            return Encoding.Unicode.GetString(head[2..]);
        }

        if (head.Length >= 2 && head[0] == 0xFE && head[1] == 0xFF)
        {
            return Encoding.BigEndianUnicode.GetString(head[2..]);
        }

        ReadOnlySpan<byte> body = head.Length >= 3 && head[0] == 0xEF && head[1] == 0xBB && head[2] == 0xBF ? head[3..] : head;
        return Encoding.UTF8.GetString(body);
    }

    // JSON, JSON lines and markup are text with commas in it; they are not tables.
    private static bool LooksLikeMarkup(string text)
    {
        foreach (char c in text)
        {
            if (char.IsWhiteSpace(c))
            {
                continue;
            }

            return c is '{' or '[' or '<';
        }

        return true;
    }

    private static bool HasEqualCounts(string text, char delimiter, bool windowCut)
    {
        List<int> counts = [];
        int count = 0;
        bool inQuotes = false;
        bool rowHasContent = false;
        bool rowOpen = false;

        for (int i = 0; i < text.Length && counts.Count < MaxRows; i++)
        {
            char c = text[i];
            rowOpen = true;
            if (c == '"')
            {
                inQuotes = !inQuotes;
                rowHasContent = true;
            }
            else if (inQuotes)
            {
                rowHasContent = true;
            }
            else if (c is '\n' or '\r')
            {
                if (c == '\r' && i + 1 < text.Length && text[i + 1] == '\n')
                {
                    i++;
                }

                if (rowHasContent)
                {
                    counts.Add(count);
                }

                count = 0;
                rowHasContent = false;
                rowOpen = false;
            }
            else
            {
                rowHasContent = true;
                if (c == delimiter)
                {
                    count++;
                }
            }
        }

        if (rowOpen && rowHasContent && !inQuotes && !windowCut && counts.Count < MaxRows)
        {
            counts.Add(count);
        }

        return counts.Count >= MinRows && counts[0] > 0 && counts.All(n => n == counts[0]);
    }
}

using System.Text.Unicode;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Text;

/// <summary>
/// Plain text: well-formed UTF-8 (a byte order mark is allowed) or UTF-16 with a byte order mark, with no control characters
/// except tab, line feed, carriage return and form feed. Text has no marker of its own, so this is the last sign to be tried:
/// whatever else the data starts like wins. A sequence cut by the end of the window is not held against the data.
/// </summary>
internal static class TextSignature
{
    private static readonly byte[] Utf8Mark = [0xEF, 0xBB, 0xBF];

    /// <summary>
    /// True when the leading bytes read as text.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        if (head.IsEmpty)
        {
            return false;
        }

        if (head.StartsWith(Utf8Mark))
        {
            return IsUtf8Text(head[Utf8Mark.Length..]);
        }

        if (head.Length >= 2 && head[0] == 0xFF && head[1] == 0xFE)
        {
            return IsUtf16Text(head[2..], littleEndian: true);
        }

        if (head.Length >= 2 && head[0] == 0xFE && head[1] == 0xFF)
        {
            return IsUtf16Text(head[2..], littleEndian: false);
        }

        return IsUtf8Text(head);
    }

    private static bool IsUtf8Text(ReadOnlySpan<byte> bytes)
    {
        ReadOnlySpan<byte> whole = bytes[..(bytes.Length - IncompleteTail(bytes))];
        if (whole.IsEmpty)
        {
            return false;
        }

        foreach (byte b in whole)
        {
            if (IsControl(b))
            {
                return false;
            }
        }

        return Utf8.IsValid(whole);
    }

    // The window may end inside a multi-byte character: up to three bytes of an unfinished sequence are left out of the check.
    private static int IncompleteTail(ReadOnlySpan<byte> bytes)
    {
        for (int back = 1; back <= Math.Min(3, bytes.Length); back++)
        {
            byte b = bytes[^back];
            if ((b & 0xC0) == 0x80)
            {
                continue;
            }

            // Only C2..F4 can start a sequence; C0, C1 and F5..FF are errors, not a cut, and are left to the validity check.
            int expected = b switch { >= 0xF0 and <= 0xF4 => 4, >= 0xE0 and < 0xF0 => 3, >= 0xC2 and < 0xE0 => 2, _ => 1 };
            return expected > back ? back : 0;
        }

        return 0;
    }

    private static bool IsUtf16Text(ReadOnlySpan<byte> bytes, bool littleEndian)
    {
        int units = bytes.Length / 2;
        if (units == 0)
        {
            return false;
        }

        for (int i = 0; i < units; i++)
        {
            char c = (char)(littleEndian ? bytes[2 * i] | (bytes[(2 * i) + 1] << 8) : (bytes[2 * i] << 8) | bytes[(2 * i) + 1]);
            if (c < 0x20 ? IsControl((byte)c) : c == 0x7F)
            {
                return false;
            }

            if (char.IsHighSurrogate(c))
            {
                if (i + 1 == units)
                {
                    break;
                }

                char next = (char)(littleEndian ? bytes[2 * (i + 1)] | (bytes[(2 * (i + 1)) + 1] << 8) : (bytes[2 * (i + 1)] << 8) | bytes[(2 * (i + 1)) + 1]);
                if (!char.IsLowSurrogate(next))
                {
                    return false;
                }

                i++;
            }
            else if (char.IsLowSurrogate(c))
            {
                return false;
            }
        }

        return true;
    }

    // Tab, line feed, form feed and carriage return are text; the rest below the space, and DEL, are not.
    private static bool IsControl(byte b)
    {
        return (b < 0x20 && b is not (0x09 or 0x0A or 0x0C or 0x0D)) || b == 0x7F;
    }
}

/// <summary>
/// The EBML header of Matroska and WebM files: its DocType names the format.
/// </summary>
internal static class EbmlHeader
{
    private static readonly byte[] Magic = [0x1A, 0x45, 0xDF, 0xA3];

    // How far after the EBML magic the DocType element is looked for.
    private const int SearchLength = 64;

    /// <summary>
    /// Returns the DocType (<c>matroska</c> or <c>webm</c>) of the file the data starts like, or <c>null</c>.
    /// </summary>
    /// <param name="head">Leading bytes of the input.</param>
    internal static string? DocType(ReadOnlySpan<byte> head)
    {
        if (head.Length < 12 || !head.StartsWith(Magic))
        {
            return null;
        }

        // DocType element: id 0x42 0x82, one-byte size (high bit set), then the text.
        ReadOnlySpan<byte> area = head[Magic.Length..Math.Min(head.Length, Magic.Length + SearchLength)];
        for (int i = 0; i + 3 <= area.Length; i++)
        {
            if (area[i] != 0x42 || area[i + 1] != 0x82 || (area[i + 2] & 0x80) == 0)
            {
                continue;
            }

            int length = area[i + 2] & 0x7F;
            if (i + 3 + length > area.Length)
            {
                return null;
            }

            ReadOnlySpan<byte> value = area.Slice(i + 3, length);
            if (value.SequenceEqual("matroska"u8))
            {
                return "matroska";
            }

            if (value.SequenceEqual("webm"u8))
            {
                return "webm";
            }
        }

        return null;
    }

    /// <summary>
    /// True when the data starts with the EBML magic.
    /// </summary>
    /// <param name="head">Leading bytes of the input.</param>
    internal static bool StartsLikeEbml(ReadOnlySpan<byte> head)
    {
        return head.StartsWith(Magic);
    }
}

using System.Text;

namespace GLAZOV.M.CONVERTER.Formats.Detection.Video;

/// <summary>
/// MOV: an ISO base media file of the brand "qt  ", or a QuickTime file without "ftyp" that starts with an atom of a known type.
/// </summary>
internal static class MovSignature
{
    // Atom types a QuickTime file without "ftyp" may start with.
    private static readonly string[] Atoms = ["moov", "mdat", "wide", "free"];

    /// <summary>
    /// True when the data starts like MOV.
    /// </summary>
    /// <param name="head">Leading bytes of the input (the detection window).</param>
    internal static bool Matches(ReadOnlySpan<byte> head)
    {
        if (head.Length < 12)
        {
            return false;
        }

        if (head[4..8].SequenceEqual("ftyp"u8))
        {
            return head[8..12].SequenceEqual("qt  "u8);
        }

        return !EbmlHeader.StartsLikeEbml(head) && !AviSignature.Matches(head) && IsAtom(head);
    }

    // Atom: four-byte big-endian size (more than the 8-byte header) and a type from the list.
    // A size spelled entirely in printable ASCII is text, not a size; when the next atom lies inside the window
    // its type must look like a type too.
    private static bool IsAtom(ReadOnlySpan<byte> head)
    {
        uint size = BinaryPrimitives.ReadUInt32BigEndian(head[..4]);
        if (size <= 8 || IsPrintable(head[..4]))
        {
            return false;
        }

        if (size <= head.Length - 8 && !IsPrintable(head.Slice((int)size + 4, 4)))
        {
            return false;
        }

        foreach (string atom in Atoms)
        {
            if (head[4..8].SequenceEqual(Encoding.ASCII.GetBytes(atom)))
            {
                return true;
            }
        }

        return false;
    }

    private static bool IsPrintable(ReadOnlySpan<byte> bytes)
    {
        foreach (byte value in bytes)
        {
            if (value is < 0x20 or > 0x7E)
            {
                return false;
            }
        }

        return true;
    }
}

using GLAZOV.M.CONVERTER.Formats.Options;

namespace GLAZOV.M.CONVERTER.Ffmpeg.Engines;

/// <summary>
/// What is done with each track: copied as it is, or encoded. The two tracks are decided separately.
/// </summary>
/// <param name="CopyVideo">The video track is copied.</param>
/// <param name="CopyAudio">The audio track is copied.</param>
internal sealed record CopyPlan(bool CopyVideo, bool CopyAudio)
{
    /// <summary>
    /// Everything is encoded.
    /// </summary>
    internal static CopyPlan EncodeAll { get; } = new(false, false);

    /// <summary>
    /// Decides per track. Video is copied when nothing asks to change it (no video settings, no actions, no
    /// <see cref="VideoOptions.Reencode"/>, and no <see cref="VideoOptions.ConvertPixelFormat"/> for an unusual format) and its codec fits the container; audio when no audio settings
    /// are set and its codec fits. Unknown codecs (<paramref name="codecs"/> is <c>null</c>) mean encoding.
    /// </summary>
    /// <param name="to">Target container.</param>
    /// <param name="video">Video settings.</param>
    /// <param name="audio">Audio settings.</param>
    /// <param name="hasOperations">Whether actions (rotate, scale) are asked for.</param>
    /// <param name="codecs">Codecs of the input tracks.</param>
    internal static CopyPlan Decide(FileFormat to, VideoOptions? video, AudioOptions? audio, bool hasOperations, TrackCodecs? codecs)
    {
        if (codecs is null || video?.Reencode == true)
        {
            return EncodeAll;
        }

        bool copyVideo = codecs.Video is not null && !hasOperations && !VideoChanged(video) && ContainerCodecs.AllowsVideo(to, codecs.Video)
            && !(video?.ConvertPixelFormat == true && IsUnusualPixelFormat(codecs.VideoPixelFormat));
        bool copyAudio = codecs.Audio is not null && !AudioChanged(audio) && ContainerCodecs.AllowsAudio(to, codecs.Audio);
        return new CopyPlan(copyVideo, copyAudio);
    }

    /// <summary>
    /// Whether the result carries no encoded track: every track the input has is copied.
    /// </summary>
    /// <param name="codecs">Codecs of the input tracks.</param>
    internal bool CopiesEverything(TrackCodecs? codecs)
    {
        return codecs is not null
            && (codecs.Video is null || CopyVideo)
            && (codecs.Audio is null || CopyAudio);
    }

    // Common 8-bit 4:2:0 (the "j" variant is the same layout with full range); an unknown format is not touched.
    private static bool IsUnusualPixelFormat(string? pixelFormat)
    {
        return pixelFormat is not null && pixelFormat is not ("yuv420p" or "yuvj420p");
    }

    private static bool VideoChanged(VideoOptions? video)
    {
        return video is { Quality: not null } or { Bitrate: not null } or { Width: not null } or { Height: not null } or { FrameRate: not null };
    }

    private static bool AudioChanged(AudioOptions? audio)
    {
        return audio is { Bitrate: not null } or { SampleRate: not null } or { Channels: not null };
    }

/// <summary>
/// FFmpeg arguments for video to video: a pure function, no process started.
/// The first video track and the first audio track (when there is one) are re-encoded with the codecs of the target container.
/// </summary>
internal static class FfmpegVideoArguments
{
    // Quality 1..100 of VideoOptions maps linearly onto the codec scale: (value at 1, value at 100).
    // A constant: this defines what the shared quality parameter means; another mapping would be another contract.
    private static readonly (int AtLowest, int AtHighest) H264Crf = (45, 12);
    private static readonly (int AtLowest, int AtHighest) Vp9Crf = (55, 15);
    private static readonly (int AtLowest, int AtHighest) Mpeg4Qscale = (31, 2);

    private enum Codec
    {
        H264,
        Vp9,
        Mpeg4,
    }

    /// <summary>
    /// Builds the arguments.
    /// </summary>
    /// <param name="to">Target container: MP4, MKV, WebM, MOV or AVI.</param>
    /// <param name="video">Video settings; only the set ones reach the arguments.</param>
    /// <param name="audio">Audio settings for the audio track.</param>
    /// <param name="inputPath">Path of the input file.</param>
    /// <param name="outputPath">Path of the result file.</param>
    /// <param name="plan">Which tracks are copied instead of encoded; <c>null</c> вЂ” both are encoded.</param>
    /// <param name="codecs">Codecs of the input tracks, when known; a copied HEVC track needs a tag for MP4 and MOV.</param>
    /// <param name="operations">Actions in order: rotate right and scale in percent; they apply after the frame size of the video settings.</param>
    /// <param name="settings">Defaults used when the options set none; <c>null</c> вЂ” the default settings.</param>
    internal static IReadOnlyList<string> Build(FileFormat to, VideoOptions? video, AudioOptions? audio, string inputPath, string outputPath, CopyPlan? plan = null, TrackCodecs? codecs = null, IEnumerable<Operation>? operations = null, FfmpegSettings? settings = null)
    {
        ArgumentNullException.ThrowIfNull(to);
        ArgumentException.ThrowIfNullOrWhiteSpace(inputPath);
        ArgumentException.ThrowIfNullOrWhiteSpace(outputPath);
        settings ??= new FfmpegSettings();

        Codec codec = VideoCodec(to);
        List<string> args =
        [
            "-hide_banner", "-loglevel", "info", "-nostats",
            "-i", inputPath,
            "-map", "0:v:0", "-map", "0:a:0?",
        ];

        if (plan?.CopyVideo == true)
        {
            args.AddRange(["-c:v", "copy"]);
            if (codecs?.Video == "hevc" && (to.Equals(FileFormat.Mp4) || to.Equals(FileFormat.Mov)))
            {
                // Matroska keeps HEVC as "hev1"; QuickTime, Safari and iOS read only "hvc1".
                args.AddRange(["-tag:v", "hvc1"]);
            }
        }
        else
        {
            args.AddRange(["-pix_fmt", "yuv420p"]);
            AddVideoCodec(args, codec, video, settings);
            AddFilters(args, codec, video, operations);
        }

        if (plan?.CopyAudio == true)
        {
            args.AddRange(["-c:a", "copy"]);
        }
        else
        {
            AddAudio(args, codec, audio, settings);
        }

        if (to.Equals(FileFormat.Mp4) || to.Equals(FileFormat.Mov))
        {
            args.AddRange(["-movflags", "+faststart"]);
        }

        args.AddRange(["-f", Muxer(to), "-y", outputPath]);
        return args;
    }

    /// <summary>
    /// Name of the video encoder for a container (also used to explain a missing encoder).
    /// </summary>
    /// <param name="to">Target container.</param>
    internal static string VideoEncoder(FileFormat to)
    {
        return VideoEncoderName(VideoCodec(to));
    }

    private static Codec VideoCodec(FileFormat to)
    {
        if (to.Equals(FileFormat.Webm))
        {
            return Codec.Vp9;
        }

        if (to.Equals(FileFormat.Avi))
        {
            return Codec.Mpeg4;
        }

        return to.Equals(FileFormat.Mp4) || to.Equals(FileFormat.Mov) || to.Equals(FileFormat.Mkv)

            : throw new ArgumentException($"Р’РёРґРµРѕ-РґРІРёР¶РѕРє РЅРµ СѓРјРµРµС‚ С„РѕСЂРјР°С‚ В«{to.Name}В».", nameof(to));
    }

    private static string Muxer(FileFormat to)
    {
        return to.Equals(FileFormat.Mkv) ? "matroska" : to.Extension;
    }

    private static void AddVideoCodec(List<string> args, Codec codec, VideoOptions? video, FfmpegSettings settings)
    {
        args.AddRange(["-c:v", VideoEncoderName(codec)]);

        if (video?.Bitrate is int bitrate)
        {
            // A bare -b:v is only a hint for these encoders and overshoots on short clips; a ceiling (VP9: a floor too) keeps it close.
            args.AddRange(["-b:v", Text(bitrate), "-maxrate", Text(bitrate)]);
            args.AddRange(codec == Codec.Vp9 ? ["-minrate", Text(bitrate)] : ["-bufsize", ((long)bitrate * 2).ToString(CultureInfo.InvariantCulture)]);
            return;
        }

        int? quality = video?.Quality;
        switch (codec)
        {
            case Codec.H264:
                args.AddRange(["-crf", Text(quality is int q ? Scale(q, H264Crf) : settings.H264Crf)]);
                break;
            case Codec.Vp9:
                // CRF mode of VP9 needs a zero bitrate; row-mt is always a gain, so it is not a setting.
                args.AddRange(["-crf", Text(quality is int v ? Scale(v, Vp9Crf) : settings.Vp9Crf), "-b:v", "0", "-row-mt", "1", "-deadline", settings.Vp9Deadline, "-cpu-used", Text(settings.Vp9CpuUsed)]);
                break;
            default:
                args.AddRange(["-q:v", Text(quality is int m ? Scale(m, Mpeg4Qscale) : settings.Mpeg4Qscale)]);
                break;
        }
    }

    private static string VideoEncoderName(Codec codec)
    {
        return codec switch
        {
            Codec.H264 => "libx264",
            Codec.Vp9 => "libvpx-vp9",
            _ => "mpeg4",
        };
    }

    // Quality 1 gives the first value, 100 the second; the scale may run down (CRF is better when smaller).
    private static int Scale(int quality, (int AtLowest, int AtHighest) range)
    {
        return (int)Math.Round(range.AtLowest + ((quality - 1) * (range.AtHighest - range.AtLowest) / 99.0));
    }

    private static void AddFilters(List<string> args, Codec codec, VideoOptions? video, IEnumerable<Operation>? operations)
    {
        List<string> filters = [];
        if (video?.Width is not null || video?.Height is not null)
        {
            string width = video.Width is int w ? Text(Even(w)) : "-2";
            string height = video.Height is int h ? Text(Even(h)) : "-2";
            filters.Add($"scale={width}:{height}");
        }
        else if (codec == Codec.H264)
        {
            // H.264 in yuv420p cannot take an odd frame size; the size is rounded down to even like a requested one.
            filters.Add("scale=trunc(iw/2)*2:trunc(ih/2)*2");
        }

        foreach (Operation operation in operations ?? [])
        {
            filters.Add(OperationFilter(operation));
        }

        if (video?.FrameRate is double rate)
        {
            filters.Add($"fps={rate.ToString("R", CultureInfo.InvariantCulture)}");
        }

        if (filters.Count > 0)
        {
            args.AddRange(["-vf", string.Join(',', filters)]);
        }
    }

    // Rotate right is a transpose; a percent scale rounds each side down to even, never below 2.
    private static string OperationFilter(Operation operation)
    {
        if (operation.Kind == OperationKind.RotateRight)
        {
            return "transpose=1";
        }

        if (operation is { Kind: OperationKind.ScalePercent, Percent: int percent })
        {
            string p = Text(percent);
            return $"scale=w='max(2,trunc(iw*{p}/200)*2)':h='max(2,trunc(ih*{p}/200)*2)'";
        }

        throw new ArgumentException($"Р”РµР№СЃС‚РІРёРµ В«{operation.Kind}В» РІРёРґРµРѕ-РґРІРёР¶РѕРє РЅРµ СѓРјРµРµС‚.", nameof(operation));
    }

    // Sizes are rounded down to even numbers, never below 2.
    private static int Even(int value)
    {
        return Math.Max(2, value - (value % 2));
    }

    private static void AddAudio(List<string> args, Codec codec, AudioOptions? audio, FfmpegSettings settings)
    {
        switch (codec)
        {
            case Codec.H264:
                args.AddRange(["-c:a", "aac", "-b:a", Text(audio?.Bitrate ?? settings.AacBitrate)]);
                break;
            case Codec.Vp9:
                args.AddRange(["-c:a", "libopus"]);
                if (audio?.Bitrate is int opusBitrate)
                {
                    args.AddRange(["-b:a", Text(opusBitrate)]);
                }

                break;
            default:
                args.AddRange(["-c:a", "libmp3lame"]);
                args.AddRange(audio?.Bitrate is int mp3Bitrate ? ["-b:a", Text(mp3Bitrate)] : ["-q:a", Text(settings.Mp3Quality)]);
                break;
        }

        if (audio?.SampleRate is int sampleRate)
        {
            args.AddRange(["-ar", Text(sampleRate)]);
        }

        if (audio?.Channels is int channels)
        {
            args.AddRange(["-ac", Text(channels)]);
        }
    }

    private static string Text(int value)
    {
        return value.ToString(CultureInfo.InvariantCulture);
    }

/// <summary>
/// Turns FFmpeg output lines into a growing percent: the input duration comes from the "Duration:" banner line,
/// the position from the <c>out_time_us</c> lines of <c>-progress</c>. Format-independent; thread-safe,
/// so the standard error and the standard output may feed it from different threads.
/// </summary>
internal sealed partial class FfmpegProgress
{
    private const string PositionKey = "out_time_us=";

    private readonly Action<int> _onPercent;
    private readonly object _lock = new();
    private long? _durationUs;
    private bool _durationSeen;
    private int _last;

    internal FfmpegProgress(Action<int> onPercent)
    {
        _onPercent = onPercent;
    }

    [GeneratedRegex(@"Duration:\s*(?:(\d+):(\d{2}):(\d{2})(?:\.(\d+))?|N/A)")]
    private static partial Regex DurationPattern();

    /// <summary>
    /// Feeds one line of FFmpeg output; reports 1вЂ“99 when the percent has grown.
    /// </summary>
    /// <param name="line">A line from the standard error or the standard output.</param>
    internal void ProcessLine(string line)
    {
        ArgumentNullException.ThrowIfNull(line);

        lock (_lock)
        {
            if (line.StartsWith(PositionKey, StringComparison.Ordinal))
            {
                ProcessPosition(line[PositionKey.Length..]);
            }
            else if (!_durationSeen)
            {
                ProcessDuration(line);
            }
        }
    }

    private void ProcessDuration(string line)
    {
        Match match = DurationPattern().Match(line);
        if (!match.Success)
        {
            return;
        }

        _durationSeen = true;
        if (!match.Groups[1].Success)
        {
            return;
        }

        double seconds = (long.Parse(match.Groups[1].Value, CultureInfo.InvariantCulture) * 3600)
            + (long.Parse(match.Groups[2].Value, CultureInfo.InvariantCulture) * 60)
            + long.Parse(match.Groups[3].Value, CultureInfo.InvariantCulture);
        if (match.Groups[4].Success)
        {
            seconds += double.Parse("0." + match.Groups[4].Value, CultureInfo.InvariantCulture);
        }

        long micro = (long)(seconds * 1_000_000);
        _durationUs = micro > 0 ? micro : null;
    }

    // N/A, junk and negative positions (FFmpeg prints them before the first frame) are not positions.
    private void ProcessPosition(string value)
    {
        if (_durationUs is not { } duration || !long.TryParse(value.Trim(), NumberStyles.None, CultureInfo.InvariantCulture, out long position))
        {
            return;
        }

        int percent = (int)Math.Min(99, position * 100.0 / duration);
        if (percent <= _last)
        {
            return;
        }

        _last = percent;
        _onPercent(percent);
    }

        {

                    break;
                }
            }

using GLAZOV.M.CONVERTER.Data;

/// <summary>
/// Arguments of LibreOffice for converting a table: pure functions that start no program.
/// Every run has its own profile, as in <see cref="LibreOfficeArguments"/>.
/// </summary>
internal static class TableArguments
{
    private const string TextFilter = "Text - txt - csv (StarCalc)";

    /// <summary>
    /// The arguments of a headless run.
    /// </summary>
    /// <param name="inputPath">The file to convert.</param>
    /// <param name="outputFolder">Folder for the result.</param>
    /// <param name="profileFolder">Folder of the user profile of this run only.</param>
    /// <param name="importFilter">Value for <c>--infilter</c> when the input is delimited text; <c>null</c> for a book.</param>
    /// <param name="convertTo">Value for <c>--convert-to</c>.</param>
    internal static IReadOnlyList<string> Convert(string inputPath, string outputFolder, string profileFolder, string? importFilter, string convertTo)
    {
        ArgumentException.ThrowIfNullOrWhiteSpace(inputPath);
        ArgumentException.ThrowIfNullOrWhiteSpace(outputFolder);
        ArgumentException.ThrowIfNullOrWhiteSpace(profileFolder);
        ArgumentException.ThrowIfNullOrWhiteSpace(convertTo);

        List<string> arguments =
        [
            $"-env:UserInstallation={new Uri(Path.GetFullPath(profileFolder)).AbsoluteUri}",
            "--headless",
            "--norestore",
            "--nolockcheck",
            "--nologo",
        ];

        if (importFilter is not null)
        {
            arguments.Add($"--infilter={importFilter}");
        }

        arguments.AddRange(["--convert-to", convertTo, "--outdir", outputFolder, inputPath]);
        return arguments;
    }

    /// <summary>
    /// The import filter of delimited text: UTF-8, first row is data, formulas are not evaluated (a cell that starts with an equals sign stays text).
    /// </summary>
    /// <param name="delimiter">Field separator.</param>
    /// <param name="quote">Text delimiter.</param>
    /// <param name="textColumns">Number of leading columns read as text, so that numbers keep their form; 0 reads numbers as numbers.</param>
    internal static string Import(char delimiter, char quote, int textColumns = 0)
    {
        string columns = string.Join('/', Enumerable.Range(1, textColumns).Select(column => $"{column}/2"));
        return $"{TextFilter}:{(int)delimiter},{(int)quote},76,1,{columns},0,false,true,false,false,false,,false";
    }

    /// <summary>
    /// The value of <c>--convert-to</c> for the format; delimited text writes every sheet to its own file, formulas are written as values.
    /// </summary>
    /// <param name="target">Result format.</param>
    /// <param name="delimiter">Field separator of a CSV or TSV result.</param>
    /// <param name="quote">Text delimiter of a CSV or TSV result.</param>
    /// <param name="valuesAsShown">Numbers and dates as they look in the sheet.</param>
    /// <param name="byteOrderMark">Whether LibreOffice itself writes the UTF-8 mark.</param>
    internal static string ConvertTo(FileFormat target, char delimiter, char quote, bool valuesAsShown, bool byteOrderMark)
    {
        if (target.Equals(FileFormat.Csv) || target.Equals(FileFormat.Tsv))
        {
            return $"csv:{TextFilter}:{(int)delimiter},{(int)quote},76,1,,0,false,{Flag(valuesAsShown)},false,false,false,-1,false,{Flag(byteOrderMark)}";
        }

        return target.Extension;
    }

    private static string Flag(bool value)
    {
        return value ? "true" : "false";
    }
}

/// <summary>
/// Tables into each other through LibreOffice: CSV, TSV, XLSX, XLS and ODS to any of the five (the same kind included, which changes the
/// separator or the encoding), CSV and TSV to PDF, and plain text to the five as one column.
/// A book that becomes CSV or TSV is written sheet by sheet: to a folder target every sheet is a file named after the input and the sheet;
/// to any other target the sheet of the settings is taken and a note says how many there were.
/// Delimited text is read here (encoding, separator, size) and handed over as UTF-8; formulas of a text input are never evaluated.
/// </summary>
internal sealed class TableEngine(OfficeSettings settings) : IConversionEngine
{
    private const int MaxTextColumns = 1000;

    private static readonly FileFormat[] Books = [FileFormat.Xlsx, FileFormat.Xls, FileFormat.Ods];

    private static readonly FileFormat[] Delimited = [FileFormat.Csv, FileFormat.Tsv];

    private static readonly FileFormat[] Targets = [FileFormat.Csv, FileFormat.Tsv, FileFormat.Xlsx, FileFormat.Xls, FileFormat.Ods];

    /// <inheritdoc />
    public IReadOnlyList<ToolSpec> Tools { get; } = [LibreOfficeTools.Soffice];

    /// <inheritdoc />
    public bool SupportsOperations => false;

    /// <inheritdoc />
    public IReadOnlyList<ConversionPair> Pairs { get; } = BuildPairs();

    /// <inheritdoc />
    public async Task ConvertAsync(ConversionRequest request, CancellationToken cancellationToken)
    {
        ArgumentNullException.ThrowIfNull(request);

        string work = Path.Combine(request.TempFolder, $"office-{Guid.NewGuid():N}");
        try
        {
            string output = Path.Combine(work, "out");
            string profile = Path.Combine(work, "profile");
            Directory.CreateDirectory(output);
            LibreOfficeProfile.Write(profile, settings.AllowExternalResources);

            string stem = request.InputPath is null ? "table" : Path.GetFileNameWithoutExtension(request.InputPath);
            bool text = request.From.Equals(FileFormat.Csv) || request.From.Equals(FileFormat.Tsv) || request.From.Equals(FileFormat.Text);
            (string input, string? importFilter) = text
                ? await PrepareTextAsync(request, work, stem, cancellationToken).ConfigureAwait(false)
                : (await PlaceBookAsync(request, work, stem, cancellationToken).ConfigureAwait(false), null);
            CheckBookFitsXls(request, input);
            request.ReportProgress(0);

            char delimiter = request.To.Equals(FileFormat.Tsv) ? '\t' : settings.CsvDelimiter;
            bool utf8 = Encodings.IsUtf8(settings.OutputEncoding());
            string convertTo = TableArguments.ConvertTo(request.To, delimiter, settings.CsvQuote, settings.CsvValuesAsShown, utf8 && settings.CsvOutputByteOrderMark);
            IReadOnlyList<string> arguments = TableArguments.Convert(input, output, profile, importFilter, convertTo);
            ToolResult result = await ToolRunner.RunAsync(new ToolInvocation(request.Tool!.Path!, arguments, OnOutputLine: _ => { }), cancellationToken).ConfigureAwait(false);

            string[] produced = [.. new DirectoryInfo(output).GetFiles().Where(file => file.Length > 0).OrderBy(file => file.CreationTimeUtc).ThenBy(file => file.Name, StringComparer.Ordinal).Select(file => file.FullName)];
            if (result.ExitCode != 0 || produced.Length == 0)
            {
                // LibreOffice exits with 0 when it cannot load a file; the missing result is the failure.
                string name = request.InputPath is null ? "С‚Р°Р±Р»РёС†Р°" : Path.GetFileName(request.InputPath);

            }

            if (request.To.Equals(FileFormat.Csv) || request.To.Equals(FileFormat.Tsv))

                await WriteSheetsAsync(request, produced, stem, cancellationToken).ConfigureAwait(false);

        finally
        {
            await WorkFolder.RemoveAsync(work).ConfigureAwait(false);
        }

    // Delimited text goes to LibreOffice as UTF-8 with a separator that is known and a size that fits the result.
    private async Task<(string Path, string Filter)> PrepareTextAsync(ConversionRequest request, string work, string stem, CancellationToken cancellationToken)
    {

        string path = Path.Combine(work, stem + ".csv");

        TableLimits.Check(shape, request.To);

    // A table that stays text is read column by column as text; the format string is long, so a very wide table is read as numbers, and the note says so.
    private int TextColumns(ConversionRequest request, TableShape shape)
    {
        bool toText = request.To.Equals(FileFormat.Csv) || request.To.Equals(FileFormat.Tsv);
        if (!toText || !settings.CsvKeepValuesAsText)
        {
            return 0;
        }

        if (shape.Columns > MaxTextColumns)
        {
            request.AddNote($"РІ С‚Р°Р±Р»РёС†Рµ {shape.Columns} СЃС‚РѕР»Р±С†РѕРІ (Р±РѕР»СЊС€Рµ {MaxTextColumns}): Р·РЅР°С‡РµРЅРёСЏ РїСЂРѕС‡РёС‚Р°РЅС‹ РєР°Рє С‡РёСЃР»Р° Рё РґР°С‚С‹, РІРµРґСѓС‰РёРµ РЅСѓР»Рё РјРѕРіР»Рё РїСЂРѕРїР°СЃС‚СЊ");

    // The real input file is used as it is when its name already carries the right extension; otherwise a copy is made.
    private static async Task<string> PlaceBookAsync(ConversionRequest request, string work, string stem, CancellationToken cancellationToken)
    {
        string extension = "." + request.From.Extension;
        if (request.InputPath is not null && string.Equals(Path.GetExtension(request.InputPath), extension, StringComparison.OrdinalIgnoreCase))
        {
            return request.InputPath;
        }

        string path = Path.Combine(work, stem + extension);
        await using FileStream copy = new(path, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, FileOptions.Asynchronous | FileOptions.SequentialScan);
        await request.Input.CopyToAsync(copy, cancellationToken).ConfigureAwait(false);
        return path;
    }

    // LibreOffice cuts a book that the old XLS cannot hold without a word, so the size is read before the run.
    private static void CheckBookFitsXls(ConversionRequest request, string input)
    {

        {
            TableLimits.Check(size, request.To);
        }
    }

    private async Task WriteSheetsAsync(ConversionRequest request, string[] sheets, string stem, CancellationToken cancellationToken)
    {
        if (request.OutputFolder is not null)
        {
            foreach (string sheet in sheets)
            {
                string name = Path.GetFileNameWithoutExtension(sheet) + "." + request.To.Extension;
                string path;
                try
                {
                    path = request.GetOutputFilePath(name);
                }
                catch (ArgumentException ex)
                {
                    throw ConversionException.InvalidParameters($"Р›РёСЃС‚ В«{SheetName(sheet, stem)}В» РЅРµР»СЊР·СЏ Р·Р°РїРёСЃР°С‚СЊ С„Р°Р№Р»РѕРј: {ex.Message}");
                }

                await using FileStream file = new(path, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, FileOptions.Asynchronous);
                await WriteEncodedAsync(sheet, file, cancellationToken).ConfigureAwait(false);
            }

            return;
        }

        // Only a book has sheets to choose from; a CSV, TSV or text gives the one file.
        int number = Books.Contains(request.From) ? settings.SheetNumber : 1;
        if (number > sheets.Length)
        {
            throw ConversionException.InvalidParameters($"Р’ РєРЅРёРіРµ {sheets.Length} Р»РёСЃС‚РѕРІ, Р° РІ РЅР°СЃС‚СЂРѕР№РєР°С… РІС‹Р±СЂР°РЅ Р»РёСЃС‚ {settings.SheetNumber} ({nameof(OfficeSettings.SheetNumber)}).");
        }

        string chosen = sheets[number - 1];
        if (sheets.Length > 1)
        {
            request.AddNote($"РІ РєРЅРёРіРµ {sheets.Length} Р»РёСЃС‚РѕРІ, РІР·СЏС‚ Р»РёСЃС‚ В«{SheetName(chosen, stem)}В»");
        }

        await WriteEncodedAsync(chosen, request.Output, cancellationToken).ConfigureAwait(false);
    }

    // LibreOffice names a sheet file "<input>-<sheet>.csv"; a single-sheet text keeps whatever it was given.
    private static string SheetName(string path, string stem)
    {
        string name = Path.GetFileNameWithoutExtension(path);
        return name.StartsWith(stem + "-", StringComparison.Ordinal) ? name[(stem.Length + 1)..] : name;
    }

    // LibreOffice wrote UTF-8 (with the mark when the settings ask for UTF-8 with it); any other encoding is made here.
    private async Task WriteEncodedAsync(string source, Stream destination, CancellationToken cancellationToken)
    {
        Encoding encoding = settings.OutputEncoding();
        if (Encodings.IsUtf8(encoding))
        {
            await using FileStream file = new(source, FileMode.Open, FileAccess.Read, FileShare.Read, 4096, FileOptions.Asynchronous | FileOptions.SequentialScan);
            await file.CopyToAsync(destination, cancellationToken).ConfigureAwait(false);
            return;
        }

        try
        {
            if (settings.CsvOutputByteOrderMark)
            {
                await destination.WriteAsync(encoding.GetPreamble(), cancellationToken).ConfigureAwait(false);
            }

            Encoder encoder = encoding.GetEncoder();
            using StreamReader reader = new(source, new UTF8Encoding(false), detectEncodingFromByteOrderMarks: false, 64 * 1024);
            char[] chars = new char[64 * 1024];
            byte[] bytes = new byte[encoding.GetMaxByteCount(chars.Length)];
            int read;
            while ((read = await reader.ReadAsync(chars.AsMemory(), cancellationToken).ConfigureAwait(false)) > 0)
            {
                int count = encoder.GetBytes(chars.AsSpan(0, read), bytes, flush: false);
                await destination.WriteAsync(bytes.AsMemory(0, count), cancellationToken).ConfigureAwait(false);
            }

            int tail = encoder.GetBytes([], bytes, flush: true);
            await destination.WriteAsync(bytes.AsMemory(0, tail), cancellationToken).ConfigureAwait(false);
        }
        catch (EncoderFallbackException)
        {
            throw ConversionException.InvalidParameters($"Р’ С‚Р°Р±Р»РёС†Рµ РµСЃС‚СЊ Р·РЅР°Рє, РєРѕС‚РѕСЂРѕРіРѕ РЅРµС‚ РІ РєРѕРґРёСЂРѕРІРєРµ В«{settings.CsvOutputEncoding}В» ({nameof(OfficeSettings.CsvOutputEncoding)}): РІС‹Р±РµСЂРёС‚Рµ UTF-8 РёР»Рё РґСЂСѓРіСѓСЋ РєРѕРґРёСЂРѕРІРєСѓ.");
        }
    }

    private static List<ConversionPair> BuildPairs()
    {
        List<ConversionPair> pairs = [];
        foreach (FileFormat from in Books.Concat(Delimited))
        {
            foreach (FileFormat to in Targets)
            {
                pairs.Add(new ConversionPair(from, to) { WritesSeveralFiles = Books.Contains(from) && Delimited.Contains(to) });
            }
        }

        pairs.AddRange(Delimited.Select(from => new ConversionPair(from, FileFormat.Pdf)));
        pairs.AddRange(Targets.Select(to => new ConversionPair(FileFormat.Text, to)));
        return pairs;
    }

/// <summary>
/// Reads a CSV, TSV or text input: turns it into UTF-8 (byte order mark, else valid UTF-8, else the encoding of the settings),
/// picks the field separator and measures the table. LibreOffice then gets plain UTF-8 and a known separator.
/// </summary>
internal static class TableText
{
    private const int BufferSize = 64 * 1024;
    private const int FirstRecordLimit = 1024 * 1024;

    /// <summary>
    /// Writes the text of <paramref name="inputPath"/> to <paramref name="outputPath"/> as UTF-8 without a mark and returns its shape.
    /// </summary>
    /// <param name="inputPath">The input file.</param>
    /// <param name="outputPath">Where the UTF-8 copy goes.</param>
    /// <param name="from">Format of the input: CSV, TSV or text.</param>
    /// <param name="settings">Separator, quote and the fallback encoding.</param>
    /// <param name="cancellationToken">Cancellation.</param>
    internal static async Task<TableShape> PrepareAsync(string inputPath, string outputPath, FileFormat from, OfficeSettings settings, CancellationToken cancellationToken)
    {
        try
        {
            await CopyAsync(inputPath, outputPath, new UTF8Encoding(encoderShouldEmitUTF8Identifier: false, throwOnInvalidBytes: true), cancellationToken).ConfigureAwait(false);
        }
        catch (DecoderFallbackException)
        {
            await ConvertFromFallbackAsync(inputPath, outputPath, settings, cancellationToken).ConfigureAwait(false);
        }

        char delimiter = await ChooseDelimiterAsync(outputPath, from, settings, cancellationToken).ConfigureAwait(false);
        return await MeasureAsync(outputPath, delimiter, settings.CsvQuote, cancellationToken).ConfigureAwait(false);
    }

    /// <summary>
    /// The separator to read with: the tab for TSV, the one of the settings for text, and for CSV the one of the settings
    /// when the first row has it outside quotes, otherwise a semicolon or a comma that the first row has.
    /// </summary>
    internal static char Choose(string firstRecord, FileFormat from, char configured, char quote)
    {
        if (from.Equals(FileFormat.Tsv))
        {
            return '\t';
        }

        if (!from.Equals(FileFormat.Csv))
        {
            return configured;
        }

        foreach (char candidate in new[] { configured, ';', ',' }.Distinct())
        {
            if (candidate != quote && CountOutsideQuotes(firstRecord, candidate, quote) > 0)
            {
                return candidate;
            }
        }

        return configured;
    }

    /// <summary>
    /// Rows and the widest row of a text read with the given separator and quote.
    /// </summary>
    internal static async Task<TableShape> MeasureAsync(string path, char delimiter, char quote, CancellationToken cancellationToken)
    {
        using StreamReader reader = new(path, new UTF8Encoding(false), detectEncodingFromByteOrderMarks: false, BufferSize);
        char[] buffer = new char[BufferSize];
        long rows = 0;
        int columns = 0;
        int fields = 1;
        bool inQuotes = false;
        bool pending = false;
        bool lastWasCarriageReturn = false;

        int read;
        while ((read = await reader.ReadAsync(buffer.AsMemory(), cancellationToken).ConfigureAwait(false)) > 0)
        {
            for (int i = 0; i < read; i++)
            {
                char c = buffer[i];
                if (lastWasCarriageReturn)
                {
                    lastWasCarriageReturn = false;
                    if (c == '\n')
                    {
                        continue;
                    }
                }

                if (c == quote)
                {
                    inQuotes = !inQuotes;
                    pending = true;
                }
                else if (!inQuotes && (c == '\n' || c == '\r'))
                {
                    rows++;
                    columns = Math.Max(columns, fields);
                    fields = 1;
                    pending = false;
                    lastWasCarriageReturn = c == '\r';
                }
                else
                {
                    pending = true;
                    if (!inQuotes && c == delimiter)
                    {
                        fields++;
                    }
                }
            }
        }

        if (pending)
        {
            rows++;
            columns = Math.Max(columns, fields);
        }

        return new TableShape(delimiter, rows, columns);
    }

    private static async Task<char> ChooseDelimiterAsync(string path, FileFormat from, OfficeSettings settings, CancellationToken cancellationToken)
    {
        using StreamReader reader = new(path, new UTF8Encoding(false), detectEncodingFromByteOrderMarks: false, BufferSize);
        StringBuilder record = new();
        char[] buffer = new char[BufferSize];
        bool inQuotes = false;
        int read;
        while (record.Length < FirstRecordLimit && (read = await reader.ReadAsync(buffer.AsMemory(), cancellationToken).ConfigureAwait(false)) > 0)
        {
            for (int i = 0; i < read; i++)
            {
                char c = buffer[i];
                if (c == settings.CsvQuote)
                {
                    inQuotes = !inQuotes;
                }
                else if (!inQuotes && c is '\n' or '\r')
                {
                    return Choose(record.ToString(), from, settings.CsvDelimiter, settings.CsvQuote);
                }

                record.Append(c);
            }
        }

        return Choose(record.ToString(), from, settings.CsvDelimiter, settings.CsvQuote);
    }

    private static int CountOutsideQuotes(string record, char delimiter, char quote)
    {
        int count = 0;
        bool inQuotes = false;
        foreach (char c in record)
        {
            if (c == quote)
            {
                inQuotes = !inQuotes;
            }
            else if (!inQuotes && c == delimiter)
            {
                count++;
            }
        }

        return count;
    }

    private static async Task ConvertFromFallbackAsync(string inputPath, string outputPath, OfficeSettings settings, CancellationToken cancellationToken)
    {
        try
        {
            await CopyAsync(inputPath, outputPath, settings.InputEncoding(), cancellationToken).ConfigureAwait(false);
        }
        catch (DecoderFallbackException)
        {
            throw ConversionException.InvalidParameters($"РўРµРєСЃС‚ РЅРµ С‡РёС‚Р°РµС‚СЃСЏ РЅРё РєР°Рє UTF-8, РЅРё РІ РєРѕРґРёСЂРѕРІРєРµ В«{settings.CsvInputEncoding}В» РёР· РЅР°СЃС‚СЂРѕРµРє ({nameof(OfficeSettings.CsvInputEncoding)}): СѓРєР°Р¶РёС‚Рµ РєРѕРґРёСЂРѕРІРєСѓ С„Р°Р№Р»Р°.");
        }
    }

    // A mark at the start of the file decides the encoding; without one the given encoding is used, and bad bytes throw.
    private static async Task CopyAsync(string inputPath, string outputPath, Encoding encoding, CancellationToken cancellationToken)
    {
        using StreamReader reader = new(inputPath, encoding, detectEncodingFromByteOrderMarks: true, BufferSize);
        await using StreamWriter writer = new(outputPath, append: false, new UTF8Encoding(false), BufferSize);
        char[] buffer = new char[BufferSize];
        int read;
        while ((read = await reader.ReadAsync(buffer.AsMemory(), cancellationToken).ConfigureAwait(false)) > 0)
        {
            await writer.WriteAsync(buffer.AsMemory(0, read), cancellationToken).ConfigureAwait(false);
        }
    }

namespace GLAZOV.M.CONVERTER.Pdf.Rendering;

/// <summary>
/// Draws one picture on a single PDF page of the same size (one pixel is one point).
/// </summary>
internal static class ImagePdfWriter
{
    /// <summary>
    /// Builds the PDF.
    /// </summary>
    /// <param name="picture">Bytes of the picture file.</param>
    /// <param name="limits">Limits of the engine.</param>
    /// <param name="cancellationToken">Cancellation between the steps.</param>
    /// <returns>The PDF file.</returns>
    /// <exception cref="InvalidDataException">The picture is damaged, empty or larger than the limit.</exception>
    internal static byte[] Write(byte[] picture, PdfLimits limits, CancellationToken cancellationToken)
    {
        using SKData data = SKData.CreateCopy(picture);
        using SKCodec codec = SKCodec.Create(data)
            ?? throw new InvalidDataException("The file is not a picture that can be decoded.");

        SKImageInfo info = codec.Info;
        if (info.Width <= 0 || info.Height <= 0 || (long)info.Width * info.Height > limits.MaxPixels)
        {
            throw new InvalidDataException(
                $"The picture is {info.Width}Г—{info.Height} pixels, above the limit of {limits.MaxPixels} (setting PdfSettings.MaxPixels).");
        }

        cancellationToken.ThrowIfCancellationRequested();

        SKEncodedOrigin origin = codec.EncodedOrigin;
        using SKBitmap bitmap = DecodeWhole(codec);
        cancellationToken.ThrowIfCancellationRequested();

        // The stored pixels are drawn and the orientation is applied once, by the matrix below.
        // A JPEG that needs no turning goes into the PDF as it is, without being re-encoded.
        using SKImage image = origin == SKEncodedOrigin.TopLeft
            ? SKImage.FromEncodedData(data) ?? throw new InvalidDataException("The picture could not be decoded.")
            : SKImage.FromBitmap(bitmap);

        bool swap = OrientationMatrix.SwapsSides(origin);
        float pageWidth = swap ? info.Height : info.Width;
        float pageHeight = swap ? info.Width : info.Height;

        using MemoryStream buffer = new();
        using (SKManagedWStream stream = new(buffer))
        using (SKDocument document = SKDocument.CreatePdf(stream)
            ?? throw new InvalidOperationException("Skia could not start a PDF document."))
        {
            using (SKCanvas canvas = document.BeginPage(pageWidth, pageHeight))
            {
                canvas.SetMatrix(OrientationMatrix.For(origin, info.Width, info.Height));
                canvas.DrawImage(image, 0, 0, SKSamplingOptions.Default);
            }

            document.Close();
        }

        cancellationToken.ThrowIfCancellationRequested();
        return buffer.ToArray();
    }

    // Decoding is lazy elsewhere and a cut-off file would give half a picture without an error, so the whole picture is decoded here.
    private static SKBitmap DecodeWhole(SKCodec codec)
    {
        SKImageInfo stored = codec.Info;
        SKAlphaType alpha = stored.AlphaType == SKAlphaType.Opaque ? SKAlphaType.Opaque : SKAlphaType.Premul;
        SKBitmap bitmap = new(new SKImageInfo(stored.Width, stored.Height, SKColorType.Rgba8888, alpha));
        SKCodecResult result = codec.GetPixels(bitmap.Info, bitmap.GetPixels());
        if (result != SKCodecResult.Success)
        {
            bitmap.Dispose();
            throw new InvalidDataException(result == SKCodecResult.IncompleteInput
                ? "The picture is cut off or damaged: the file ends before the picture does."
                : $"The picture could not be decoded ({result}).");
        }

        return bitmap;
    }

/// The turn that a picture's EXIF orientation asks for.

{
    /// <summary>
    /// True when the orientation swaps width and height (a quarter turn or a transpose).
    /// </summary>
    /// <param name="origin">Orientation stored in the file.</param>

    internal static byte[] Rasterize(byte[] svg, FileFormat to, ConversionOptions options, RasterLimits limits, CancellationToken cancellationToken)

        (SKEncodedImageFormat format, int quality, SKColor background) = Encoding(to, options.Image?.Quality, limits);

        using SKSvg document = new();

        document.Settings.EnableExternalJavaScript = false;

            canvas.SetMatrix(plan.Matrix);
            canvas.DrawPicture(picture);

        cancellationToken.ThrowIfCancellationRequested();

        using SKImage image = SKImage.FromBitmap(bitmap);

    {
        if (to.Equals(FileFormat.Png))
        {
            return (SKEncodedImageFormat.Png, 100, SKColors.Transparent);
        }

        if (to.Equals(FileFormat.Jpeg))
        {
            return (SKEncodedImageFormat.Jpeg, quality ?? limits.DefaultJpegQuality, limits.JpegBackground);
        }

        if (to.Equals(FileFormat.Webp))

            return (SKEncodedImageFormat.Webp, quality ?? limits.DefaultWebpQuality, SKColors.Transparent);
        }

        throw new NotSupportedException($"Р”РІРёР¶РѕРє SVG РЅРµ СѓРјРµРµС‚ С„РѕСЂРјР°С‚ СЂРµР·СѓР»СЊС‚Р°С‚Р° В«{to.Name}В».");
