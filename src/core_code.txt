using GLAZOV.M.CONVERTER.Data;
using GLAZOV.M.CONVERTER.Errors;
using GLAZOV.M.CONVERTER.Extensions;
using GLAZOV.M.CONVERTER.Formats;
using GLAZOV.M.CONVERTER.IO;
using GLAZOV.M.CONVERTER.Tooling;
using System.Text;

namespace GLAZOV.M.CONVERTER.Office.Engines;

/// <summary>
/// Tables into each other through LibreOffice: CSV, TSV, XLSX, XLS and ODS to any of the five (the same kind included, which changes the
/// separator or the encoding), CSV and TSV to PDF, and plain text to the five as one column.
/// A book that becomes CSV or TSV is written sheet by sheet: to a folder target every sheet is a file named after the input and the sheet;
/// to any other target the sheet of the settings is taken and a note says how many there were.
/// Delimited text is read here (encoding, separator, size) and handed over as UTF-8; formulas of a text input are never evaluated.
/// </summary>
internal sealed class TableEngine(OfficeSettings settings) : IConversionEngine
{
    private const int MaxTextColumns = 1000;

    private static readonly FileFormat[] Books = [FileFormat.Xlsx, FileFormat.Xls, FileFormat.Ods];

    private static readonly FileFormat[] Delimited = [FileFormat.Csv, FileFormat.Tsv];

    private static readonly FileFormat[] Targets = [FileFormat.Csv, FileFormat.Tsv, FileFormat.Xlsx, FileFormat.Xls, FileFormat.Ods];

    /// <inheritdoc />
    public IReadOnlyList<ToolSpec> Tools { get; } = [LibreOfficeTools.Soffice];

    /// <inheritdoc />
    public bool SupportsOperations => false;

    /// <inheritdoc />
    public IReadOnlyList<ConversionPair> Pairs { get; } = BuildPairs();

    /// <inheritdoc />
    public async Task ConvertAsync(ConversionRequest request, CancellationToken cancellationToken = default)
    {
        ArgumentNullException.ThrowIfNull(request);

        string work = Path.Combine(request.TempFolder, $"office-{Guid.NewGuid():N}");
        try
        {
            string output = Path.Combine(work, "out");
            string profile = Path.Combine(work, "profile");
            Directory.CreateDirectory(output);
            LibreOfficeProfile.Write(profile, settings.AllowExternalResources);

            string stem = request.InputPath is null ? "table" : Path.GetFileNameWithoutExtension(request.InputPath);
            bool text = request.From.Equals(FileFormat.Csv) || request.From.Equals(FileFormat.Tsv) || request.From.Equals(FileFormat.Text);
            (string input, string? importFilter) = text
                ? await PrepareTextAsync(request, work, stem, cancellationToken).ConfigureAwait(false)
                : (await PlaceBookAsync(request, work, stem, cancellationToken).ConfigureAwait(false), null);
            CheckBookFitsXls(request, input);
            request.ReportProgress(0);

            char delimiter = request.To.Equals(FileFormat.Tsv) ? '\t' : settings.CsvDelimiter;
            bool utf8 = Encodings.IsUtf8(settings.OutputEncoding());
            string convertTo = TableArguments.ConvertTo(request.To, delimiter, settings.CsvQuote, settings.CsvValuesAsShown, utf8 && settings.CsvOutputByteOrderMark);
            IReadOnlyList<string> arguments = TableArguments.Convert(input, output, profile, importFilter, convertTo);
            ToolResult result = await ToolRunner.RunAsync(new ToolInvocation(request.Tool!.Path!, arguments, OnOutputLine: _ => { }), cancellationToken).ConfigureAwait(false);

            string[] produced = [.. new DirectoryInfo(output).GetFiles().Where(file => file.Length > 0).OrderBy(file => file.CreationTimeUtc).ThenBy(file => file.Name, StringComparer.Ordinal).Select(file => file.FullName)];
            if (result.ExitCode != 0 || produced.Length == 0)
            {
                // LibreOffice exits with 0 when it cannot load a file; the missing result is the failure.
                string name = request.InputPath is null ? "таблица" : Path.GetFileName(request.InputPath);
                throw ConversionException.ToolFailed(request.Tool.Name, result.ExitCode, $"{result.ErrorOutput}\nLibreOffice не смог открыть файл «{name}»".TrimStart());
            }

            if (request.To.Equals(FileFormat.Csv) || request.To.Equals(FileFormat.Tsv))
            {
                await WriteSheetsAsync(request, produced, stem, cancellationToken).ConfigureAwait(false);
            }
            else
            {
                await using FileStream file = new(produced[0], FileMode.Open, FileAccess.Read, FileShare.Read, 4096, FileOptions.Asynchronous | FileOptions.SequentialScan);
                await file.CopyToAsync(request.Output, cancellationToken).ConfigureAwait(false);
            }
        }
        finally
        {
            await WorkFolder.RemoveAsync(work).ConfigureAwait(false);
        }
    }

    // Delimited text goes to LibreOffice as UTF-8 with a separator that is known and a size that fits the result.
    private async Task<(string Path, string Filter)> PrepareTextAsync(ConversionRequest request, string work, string stem, CancellationToken cancellationToken)
    {
        await using InputFile file = await request.EnsureInputFileAsync(cancellationToken).ConfigureAwait(false);
        string path = Path.Combine(work, stem + ".csv");
        TableShape shape = await TableText.PrepareAsync(file.Path, path, request.From, settings, cancellationToken).ConfigureAwait(false);
        TableLimits.Check(shape, request.To);
        return (path, TableArguments.Import(shape.Delimiter, settings.CsvQuote, TextColumns(request, shape)));
    }

    // A table that stays text is read column by column as text; the format string is long, so a very wide table is read as numbers, and the note says so.
    private int TextColumns(ConversionRequest request, TableShape shape)
    {
        bool toText = request.To.Equals(FileFormat.Csv) || request.To.Equals(FileFormat.Tsv);
        if (!toText || !settings.CsvKeepValuesAsText)
        {
            return 0;
        }

        if (shape.Columns > MaxTextColumns)
        {
            request.AddNote($"в таблице {shape.Columns} столбцов (больше {MaxTextColumns}): значения прочитаны как числа и даты, ведущие нули могли пропасть");
            return 0;
        }

        return shape.Columns;
    }

    // The real input file is used as it is when its name already carries the right extension; otherwise a copy is made.
    private static async Task<string> PlaceBookAsync(ConversionRequest request, string work, string stem, CancellationToken cancellationToken)
    {
        string extension = "." + request.From.Extension;
        if (request.InputPath is not null && string.Equals(Path.GetExtension(request.InputPath), extension, StringComparison.OrdinalIgnoreCase))
        {
            return request.InputPath;
        }

        string path = Path.Combine(work, stem + extension);
        await using FileStream copy = new(path, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, FileOptions.Asynchronous | FileOptions.SequentialScan);
        await request.Input.CopyToAsync(copy, cancellationToken).ConfigureAwait(false);
        return path;
    }

    // LibreOffice cuts a book that the old XLS cannot hold without a word, so the size is read before the run.
    private static void CheckBookFitsXls(ConversionRequest request, string input)
    {
        if (request.To.Equals(FileFormat.Xls) && BookSize.Measure(input, request.From) is { } size)
        {
            TableLimits.Check(size, request.To);
        }
    }

    private async Task WriteSheetsAsync(ConversionRequest request, string[] sheets, string stem, CancellationToken cancellationToken)
    {
        if (request.OutputFolder is not null)
        {
            foreach (string sheet in sheets)
            {
                string name = Path.GetFileNameWithoutExtension(sheet) + "." + request.To.Extension;
                string path;
                try
                {
                    path = request.GetOutputFilePath(name);
                }
                catch (ArgumentException ex)
                {
                    throw ConversionException.InvalidParameters($"Лист «{SheetName(sheet, stem)}» нельзя записать файлом: {ex.Message}");
                }

                await using FileStream file = new(path, FileMode.CreateNew, FileAccess.Write, FileShare.None, 4096, FileOptions.Asynchronous);
                await WriteEncodedAsync(sheet, file, cancellationToken).ConfigureAwait(false);
            }

            return;
        }

        // Only a book has sheets to choose from; a CSV, TSV or text gives the one file.
        int number = Books.Contains(request.From) ? settings.SheetNumber : 1;
        if (number > sheets.Length)
        {
            throw ConversionException.InvalidParameters($"В книге {sheets.Length} листов, а в настройках выбран лист {settings.SheetNumber} ({nameof(OfficeSettings.SheetNumber)}).");
        }

        string chosen = sheets[number - 1];
        if (sheets.Length > 1)
        {
            request.AddNote($"в книге {sheets.Length} листов, взят лист «{SheetName(chosen, stem)}»");
        }

        await WriteEncodedAsync(chosen, request.Output, cancellationToken).ConfigureAwait(false);
    }

    // LibreOffice names a sheet file "<input>-<sheet>.csv"; a single-sheet text keeps whatever it was given.
    private static string SheetName(string path, string stem)
    {
        string name = Path.GetFileNameWithoutExtension(path);
        return name.StartsWith(stem + "-", StringComparison.Ordinal) ? name[(stem.Length + 1)..] : name;
    }

    // LibreOffice wrote UTF-8 (with the mark when the settings ask for UTF-8 with it); any other encoding is made here.
    private async Task WriteEncodedAsync(string source, Stream destination, CancellationToken cancellationToken)
    {
        Encoding encoding = settings.OutputEncoding();
        if (Encodings.IsUtf8(encoding))
        {
            await using FileStream file = new(source, FileMode.Open, FileAccess.Read, FileShare.Read, 4096, FileOptions.Asynchronous | FileOptions.SequentialScan);
            await file.CopyToAsync(destination, cancellationToken).ConfigureAwait(false);
            return;
        }

        try
        {
            if (settings.CsvOutputByteOrderMark)
            {
                await destination.WriteAsync(encoding.GetPreamble(), cancellationToken).ConfigureAwait(false);
            }

            Encoder encoder = encoding.GetEncoder();
            using StreamReader reader = new(source, new UTF8Encoding(false), detectEncodingFromByteOrderMarks: false, 64 * 1024);
            char[] chars = new char[64 * 1024];
            byte[] bytes = new byte[encoding.GetMaxByteCount(chars.Length)];
            int read;
            while ((read = await reader.ReadAsync(chars.AsMemory(), cancellationToken).ConfigureAwait(false)) > 0)
            {
                int count = encoder.GetBytes(chars.AsSpan(0, read), bytes, flush: false);
                await destination.WriteAsync(bytes.AsMemory(0, count), cancellationToken).ConfigureAwait(false);
            }

            int tail = encoder.GetBytes([], bytes, flush: true);
            await destination.WriteAsync(bytes.AsMemory(0, tail), cancellationToken).ConfigureAwait(false);
        }
        catch (EncoderFallbackException)
        {
            throw ConversionException.InvalidParameters($"В таблице есть знак, которого нет в кодировке «{settings.CsvOutputEncoding}» ({nameof(OfficeSettings.CsvOutputEncoding)}): выберите UTF-8 или другую кодировку.");
        }
    }

    private static List<ConversionPair> BuildPairs()
    {
        List<ConversionPair> pairs = [];
        foreach (FileFormat from in Books.Concat(Delimited))
        {
            foreach (FileFormat to in Targets)
            {
                pairs.Add(new ConversionPair(from, to) { WritesSeveralFiles = Books.Contains(from) && Delimited.Contains(to) });
            }
        }

        pairs.AddRange(Delimited.Select(from => new ConversionPair(from, FileFormat.Pdf)));
        pairs.AddRange(Targets.Select(to => new ConversionPair(FileFormat.Text, to)));
        return pairs;
    }
}

using System.Text;
using GLAZOV.M.CONVERTER.Converting;
using GLAZOV.M.CONVERTER.Data;
using GLAZOV.M.CONVERTER.Errors;
using GLAZOV.M.CONVERTER.Formats;
using GLAZOV.M.CONVERTER.Office.Engines;
using GLAZOV.M.CONVERTER.Settings;

namespace GLAZOV.M.CONVERTER.Office.Tests.Engines;

/// <summary>
/// Tables into each other through the real LibreOffice (skipped without it): a book of three sheets to a folder and to one file,
/// text in several encodings there and back, formulas that stay text, the limits of the old XLS, text as a one-column table, a table as PDF.
/// </summary>
[Collection("LibreOffice")]
public sealed class TableEngineTests : IDisposable
{
    private static readonly Encoding Cp1251 = Encodings.Find("windows-1251", "test");

    private readonly string _temp = Path.Combine(Path.GetTempPath(), "convtable-" + Guid.NewGuid().ToString("N"));

    public TableEngineTests()
    {
        Directory.CreateDirectory(_temp);
    }

    public void Dispose()
    {
        Directory.Delete(_temp, recursive: true);
    }

    public static TheoryData<string> Books => ["xlsx", "xls", "ods"];

    private Converter Make(OfficeSettings? office = null, bool overwrite = false)
    {
        string folder = Path.Combine(_temp, "work");
        Directory.CreateDirectory(folder);
        ConverterSettings settings = new() { TempFolder = folder, DetectionWindow = 4 * 1024 * 1024, OverwriteExisting = overwrite };
        return OfficeConverter.Create(settings, office);
    }

    private static string Book(string extension)
    {
        return Path.Combine(AppContext.BaseDirectory, "Samples", "sample-book." + extension);
    }

    private string Put(string name, byte[] bytes)
    {
        string path = Path.Combine(_temp, "in", name);
        Directory.CreateDirectory(Path.GetDirectoryName(path)!);
        File.WriteAllBytes(path, bytes);
        return path;
    }

    private string Out => Path.Combine(_temp, "out");

    private string[] LeftInWork()
    {
        string folder = Path.Combine(_temp, "work");
        return Directory.Exists(folder) ? Directory.GetFileSystemEntries(folder) : [];
    }

    private static string Utf8(byte[] bytes)
    {
        return new UTF8Encoding(false).GetString(bytes.AsSpan().StartsWith(Encoding.UTF8.GetPreamble()) ? bytes[3..] : bytes);
    }

    [OfficeTheory]
    [MemberData(nameof(Books))]
    public async Task Book_ToAFolder_GivesAFilePerSheet_NamedAfterTheInputAndTheSheet(string extension)
    {
        ConversionResult result = await Make().ConvertAsync(Source.File(Book(extension)), Target.Folder(Out, FileFormat.Csv));

        string[] names = [.. result.OutputPaths.Select(p => Path.GetFileName(p)!).Order(StringComparer.Ordinal)];
        Assert.Equal(["sample-book-Alpha.csv", "sample-book-Zeta.csv", "sample-book-Итог.csv"], names);
        Assert.Equal(names, Directory.GetFiles(Out).Select(Path.GetFileName).Order(StringComparer.Ordinal));
        Assert.Equal("a1", Utf8(File.ReadAllBytes(Path.Combine(Out, "sample-book-Alpha.csv"))).Trim());
        Assert.Contains("\"a,b \"\"q\"\"\"", Utf8(File.ReadAllBytes(Path.Combine(Out, "sample-book-Итог.csv"))));
        Assert.Contains(Path.Combine(Path.GetFullPath(Out), "sample-book-Итог.csv"), result.OutputPaths);
        Assert.Empty(LeftInWork());
    }

    [OfficeFact]
    public async Task Book_ToTsvFolder_UsesTabsAndTheTsvExtension()
    {
        ConversionResult result = await Make().ConvertAsync(Source.File(Book("xlsx")), Target.Folder(Out, FileFormat.Tsv));

        Assert.All(result.OutputPaths, path => Assert.EndsWith(".tsv", path, StringComparison.Ordinal));
        Assert.Contains("Имя\t", Utf8(File.ReadAllBytes(Path.Combine(Out, "sample-book-Итог.tsv"))));
    }

    [OfficeFact]
    public async Task Book_ToOneFile_TakesTheFirstSheet_StartsWithTheMark_AndSaysSo()
    {
        string output = Path.Combine(Out, "one.csv");

        ConversionResult result = await Make().ConvertAsync(Source.File(Book("xlsx")), Target.File(output, FileFormat.Csv));

        byte[] bytes = File.ReadAllBytes(output);
        Assert.True(bytes.AsSpan().StartsWith(Encoding.UTF8.GetPreamble()));
        Assert.StartsWith("Имя,", Utf8(bytes));
        Assert.Equal([output], result.OutputPaths);
        Assert.Equal(["в книге 3 листов, взят лист «Итог»"], result.Notes);
    }

    [OfficeFact]
    public async Task Book_ToMemory_SheetNumberFromTheSettings_NoMarkWhenOff()
    {
        OfficeSettings office = new() { SheetNumber = 2, CsvOutputByteOrderMark = false };

        ConversionResult result = await Make(office).ConvertAsync(Source.File(Book("ods")), Target.Memory(FileFormat.Csv));

        Assert.Equal("z1", Utf8(result.Bytes!).Trim());
        Assert.False(result.Bytes.AsSpan().StartsWith(Encoding.UTF8.GetPreamble()));
        Assert.Contains("«Zeta»", result.Notes.Single());
    }

    [OfficeFact]
    public async Task Book_SheetNumberBeyondTheBook_IsInvalidParameters()
    {
        ConversionException error = await Assert.ThrowsAsync<ConversionException>(
            () => Make(new OfficeSettings { SheetNumber = 4 }).ConvertAsync(Source.File(Book("xlsx")), Target.Memory(FileFormat.Csv)));

        Assert.Equal(ConversionFailureReason.InvalidParameters, error.Reason);
        Assert.Contains("3 листов", error.Message);
        Assert.Empty(LeftInWork());
    }

    [OfficeFact]
    public async Task Book_ExistingFileInTheFolder_IsNotOverwritten_AndNoSheetIsWritten()
    {
        Directory.CreateDirectory(Out);
        File.WriteAllText(Path.Combine(Out, "sample-book-Zeta.csv"), "mine");

        ConversionException error = await Assert.ThrowsAsync<ConversionException>(
            () => Make().ConvertAsync(Source.File(Book("xlsx")), Target.Folder(Out, FileFormat.Csv)));

        Assert.Equal(ConversionFailureReason.FileExists, error.Reason);
        Assert.Equal(["sample-book-Zeta.csv"], Directory.GetFileSystemEntries(Out).Select(Path.GetFileName));
    }

    [OfficeFact]
    public async Task Book_ToBook_KeepsTheSheets()
    {
        string output = Path.Combine(Out, "copy.ods");

        await Make().ConvertAsync(Source.File(Book("xlsx")), Target.File(output, FileFormat.Ods));
        ConversionResult back = await Make().ConvertAsync(Source.File(output), Target.Memory(FileFormat.Csv));

        Assert.Equal(3, int.Parse(back.Notes.Single().Split(' ')[2], System.Globalization.CultureInfo.InvariantCulture));
    }

    [OfficeFact]
    public async Task Csv_ToWindows1251_AndSemicolons_WritesThoseBytes()
    {
        string text = "Имя;Город\r\nИван;Москва\r\nПётр;Казань\r\n";
        string source = Put("люди.csv", Encoding.UTF8.GetBytes(text));
        string output = Path.Combine(Out, "b.csv");

        await Make(new OfficeSettings { CsvDelimiter = ';', CsvOutputEncoding = "windows-1251" }).ConvertAsync(Source.File(source), Target.File(output, FileFormat.Csv));

        Assert.Equal(Cp1251.GetBytes(text), File.ReadAllBytes(output));
    }

    [OfficeFact]
    public async Task Csv_Utf8WithAndWithoutMark_ReadTheSame()
    {
        string text = "Имя,Город\nИван,Москва\nПётр,Казань\n";
        string plain = Put("a.csv", Encoding.UTF8.GetBytes(text));
        string marked = Put("b.csv", [.. Encoding.UTF8.GetPreamble(), .. Encoding.UTF8.GetBytes(text)]);

        ConversionResult first = await Make().ConvertAsync(Source.File(plain), Target.Memory(FileFormat.Csv));
        ConversionResult second = await Make().ConvertAsync(Source.File(marked), Target.Memory(FileFormat.Csv));

        Assert.Equal(Utf8(first.Bytes!), Utf8(second.Bytes!));
        Assert.Contains("Москва", Utf8(first.Bytes!));
    }

    [OfficeFact]
    public async Task Csv_WithSemicolons_AndTheDefaultSeparator_IsReadBySemicolons_WrittenWithCommas()
    {
        string source = Put("s.csv", Encoding.UTF8.GetBytes("a;b;c\n1;2;3\n4;5;6\n"));

        ConversionResult result = await Make().ConvertAsync(Source.File(source), Target.Memory(FileFormat.Csv));

        Assert.Equal("a,b,c\r\n1,2,3\r\n4,5,6", Utf8(result.Bytes!).Trim());
    }

    [OfficeFact]
    public async Task CsvToTsv_AndBack_ChangesOnlyTheSeparator()
    {
        string source = Put("c.csv", Encoding.UTF8.GetBytes("a,b\n\"x,y\",2\n3,4\n"));
        string tsv = Path.Combine(Out, "c.tsv");
        string csv = Path.Combine(Out, "c2.csv");

        await Make().ConvertAsync(Source.File(source), Target.File(tsv, FileFormat.Tsv));
        await Make().ConvertAsync(Source.File(tsv), Target.File(csv, FileFormat.Csv));

        Assert.Equal("a\tb\r\nx,y\t2\r\n3\t4", Utf8(File.ReadAllBytes(tsv)).Trim());
        Assert.Equal("a,b\r\n\"x,y\",2\r\n3,4", Utf8(File.ReadAllBytes(csv)).Trim());
    }

    [OfficeFact]
    public async Task FormulaInAnInputCsv_StaysText()
    {
        string source = Put("f.csv", Encoding.UTF8.GetBytes("a,b\n=1+1,@SUM(1)\n3,4\n"));

        ConversionResult result = await Make().ConvertAsync(Source.File(source), Target.Memory(FileFormat.Csv));

        Assert.Contains("=1+1", Utf8(result.Bytes!));
        Assert.DoesNotContain("\r\n2,", Utf8(result.Bytes!));
    }

    [OfficeFact]
    public async Task CharacterTheOutputEncodingCannotHold_IsInvalidParameters()
    {
        string source = Put("j.csv", Encoding.UTF8.GetBytes("a,b\nこんにちは,2\n3,4\n"));

        ConversionException error = await Assert.ThrowsAsync<ConversionException>(
            () => Make(new OfficeSettings { CsvOutputEncoding = "windows-1251" }).ConvertAsync(Source.File(source), Target.Memory(FileFormat.Csv)));

        Assert.Equal(ConversionFailureReason.InvalidParameters, error.Reason);
        Assert.Contains("windows-1251", error.Message);
    }

    [OfficeFact]
    public async Task CsvTooTallForXls_IsRefusedWithTheLimit_NotCut()
    {
        string source = Put("tall.csv", Encoding.UTF8.GetBytes(string.Concat(Enumerable.Repeat("a,b\n", 65_537))));

        ConversionException error = await Assert.ThrowsAsync<ConversionException>(
            () => Make().ConvertAsync(Source.File(source), Target.File(Path.Combine(Out, "t.xls"), FileFormat.Xls)));

        Assert.Equal(ConversionFailureReason.InvalidParameters, error.Reason);
        Assert.Contains("65536", error.Message);
        Assert.False(File.Exists(Path.Combine(Out, "t.xls")));
        Assert.Empty(LeftInWork());
    }

    [OfficeFact]
    public async Task Text_ToXlsx_IsOneColumn_AndBackToCsv()
    {
        string source = Put("заметка.txt", Encoding.UTF8.GetBytes("первая строка\nвторая строка\nтретья строка\n"));
        string xlsx = Path.Combine(Out, "t.xlsx");

        await Make().ConvertAsync(Source.File(source), Target.File(xlsx, FileFormat.Xlsx));
        ConversionResult back = await Make().ConvertAsync(Source.File(xlsx), Target.Memory(FileFormat.Csv));

        Assert.Equal("первая строка\r\nвторая строка\r\nтретья строка", Utf8(back.Bytes!).Trim());
    }

    [OfficeFact]
    public async Task Csv_ToPdf_IsAPdf()
    {
        string source = Put("p.csv", Encoding.UTF8.GetBytes("Имя,Город\nИван,Москва\nПётр,Казань\n"));
        string pdf = Path.Combine(Out, "p.pdf");

        await Make().ConvertAsync(Source.File(source), Target.File(pdf, FileFormat.Pdf));

        Assert.StartsWith("%PDF", Encoding.ASCII.GetString(File.ReadAllBytes(pdf), 0, 4), StringComparison.Ordinal);
        Assert.Empty(LeftInWork());
    }

    [OfficeFact]
    public async Task Stream_Input_IsAccepted_AndLeavesNothingBehind()
    {
        await using MemoryStream input = new(Encoding.UTF8.GetBytes("a,b\n1,2\n3,4\n"));

        ConversionResult result = await Make().ConvertAsync(Source.Stream(input), Target.Memory(FileFormat.Tsv));

        Assert.Equal("a\tb\r\n1\t2\r\n3\t4", Utf8(result.Bytes!).Trim());
        Assert.Empty(LeftInWork());
    }

    [OfficeFact]
    public async Task Batch_BooksToAFolder_ReportsEveryFile()
    {
        string folder = Path.Combine(_temp, "books");
        Directory.CreateDirectory(folder);
        File.Copy(Book("xlsx"), Path.Combine(folder, "one.xlsx"));
        File.Copy(Book("ods"), Path.Combine(folder, "two.ods"));

        GLAZOV.M.CONVERTER.Batch.BatchReport report = await Make().ConvertBatchAsync(Source.Folder(folder), Target.Folder(Out, FileFormat.Csv));

        Assert.Equal(2, report.Succeeded);
        Assert.Equal(6, report.Items.Sum(item => item.OutputPaths.Count));
        Assert.Equal(6, Directory.GetFiles(Out).Length);
    }
}
